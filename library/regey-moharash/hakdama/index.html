הקדמה

תוכן

מִי יְמַלֵּל גְּבוּרוֹת הֲוָיָ"ה יַשְׁמִיעַ כָּל תְּהִלָּתוֹ. יַעַן שֶׁזִּכָּנִי הקב"ה בִּזְכֻיּוֹת חִנָּם לְהִתְוַדַּע מֵרַבֵּנוּ זַ"ל, וְזֶה לִי אַרְבָּעִים שָׁנָה, שֶׁאֲנִי עוֹסֵק בְּצָרְכֵי צִבּוּר לְגַלּוֹת וּלְפַרְסֵם אֶת לִמּוּדוֹ הֶעָמקֹ וְהַתָּמִים שֶׁל רַבֵּנוּ זַ"ל בָּעוֹלָם, וְרַבּוֹת עָבְרוּ עָלַי בְּמֶשֶׁךְ הַשָּׁנִים הָאֵלּוּ, כָּל מִינֵי עֲלִיּוֹת וִירִידוֹת, גְּבוּרוֹת וְדִינִים קָשִׁים, מַחֲלֹקֶת עֶלְבּוֹנוֹת וּבִזְיוֹנוֹת הָיוּ מְנַת חֶלְקִי, וְזִכָּנִי הקב"ה לְחַבֵּר הֲמוֹן סְפָרִים, אָמַרְתִּי לִרְשֹׁם עַל פְּנֵי הַכְּתָב אֱמֶת מְעַט מִזְעֵיר כְּטִפָּה מִן הַיָּם מֶה עָבַר עָלַי, כְּדֵי שֶׁיַּעַמְדוּ לְיָמִים רַבִּים לִפְנֵי בָּנַי וְתַלְמִידַי הָרַבִּים, שֶׁזָּכוּ לְהִתְקָרֵב וַעֲדַיִן יוֹנְקִים וְשׁוֹתִים וְנֶהֱנִים מִסְּפָרַי הָרַבִּים, שֶׁזָּכִיתִי לְחַדֵּשׁ בְּדִבְרֵי רַבֵּנוּ זַ"ל, כִּיּ אֲנִי זָקַנְתִּי וְהִגַּעְתִּי לִימֵי זִקְנָה, וְרַבּוֹת עָבְרוּ עָלַי, וּמִי יוֹדֵעַ מַה יֵּלֵד יוֹם, אָמַרְתִּי יָמִים יְדַבְּרוּ, וּדְבָרַי אֵלּוּ עוֹד יִהְיוּ תּוֹעֶלֶת גְּדוֹלָה לַמְּעַיְּנִים בָּהֶם.

מָה אָשִׁיב לַהֲוָיָ"ה כָּל תַּגְמוּלוֹהִי עָלַי

אֵין מִלִּים בְּפִי לְהוֹדוֹת וּלְהַלֵּל וּלְשַׁבֵּחַ אוֹתְךָ תִּתְבָּרַךְ עַל כָּל הַחֶסֶד חִנָּם שֶׁעָשִׂיתָ עִמָּדִי, אֲשֶׁר מִנְּעוּרַי בִּהְיוֹתִי בְּגִיל ט"ז זָכִיתִי לְהִתְוַדַּע מֵהָאוֹר הָאוֹרוֹת, צַח הַצַּחְצָחוֹת, רַבֵּנוּ הַקָּדוֹשׁ וְהַנּוֹרָא רַבִּי נַחְמָן מִבְּרֶסְלֶב, זְכוּתוֹ תָּגֵן עָלֵינוּ, וְלֹא עוֹד שֶׁהִגְדַּלְתָּ עָלַי חֲסָדֶיךָ, שֶׁזָּכִיתִי לִהְיוֹת שַׁמָּשׁ אֵצֶל רַבֵּנוּ זַ"ל, וְזֶה לִי אַרְבָּעִים שָׁנִים מִשְּׁנַת תשכ"ד עַד שְׁנַת תשס"ד, תַּאֲרִיךְ עוֹד יָמַי וּשְׁנוֹתַי עַל פְּנֵי אֲדָמָה, שֶׁזָּכִיתִי לַעֲסקֹ בְּגִלּוּי וּפִרְסוּם שֵׁם הַצַּדִּיק הָאֱמֶת בָּעוֹלָם, יוֹמָם וָלַיְלָה בְּלִי שׁוּם גֻּזְמָא כְּלָל, אַרְבָּעִים שָׁנָה מְלֵאִים וּשְׁלֵמִים הָיוּ, שֶׁמָּסַרְתִּי אֶת נַפְשִׁי וּשְׁמִי וְכָל מַה שֶּׁהָיָה לִי, הַכֹּל לַכֹּל רַק בַּעֲבוּר לְגַלּוֹת וּלְפַרְסֵם אֶת רַבֵּנוּ זַ"ל בָּעוֹלָם, רַבּוֹת עָבְרוּ עָלַי, אֲשֶׁר אִם אֶרְצֶה לְרָשְׁמָם עַל פְּנֵי הַכְּתָב, יִכְלוּ כָּל הַיְרִיעוֹת, סָבַלְתִּי עֶלְבּוֹנוֹת, חֵרוּפִין וְגִדּוּפִין, חֲרָפוֹת וּבִזְיוֹנוֹת וּשְׁפִיכוּת דָּמִים, וּבַעֲבוּר כָּל זֶה הֶחְרִימוּ אוֹתִי, וּבָדְלוּ מִמֶּנִּי יְדִידַי הַטּוֹבִים. וּמַה לֹּא עָבַר עָלַי, וַאֲנִי עָשִׂיתִי עַצְמִי כְּאַסְקֻפָּה הַנִּדְרֶסֶת לַכֹּל, וְאֶת כָּל כֻּלִּי מָסַרְתִּי עַצְמִי, רַק כְּדֵי לְגַלּוֹת וּלְפַרְסֵם אֶת שֵׁם הַצַּדִּיק בָּעוֹלָם, וְחֵפֶץ ה' בְּיָדִי הִצְלִיחַ, אַחֲרֵי עֲבוֹדָה וִיגִיעָה וְטִרְחָה שֶׁל אַרְבָּעִים שָׁנִים רְצוּפוֹת וּמְסִירוּת נֶפֶשׁ מַמָּשׁ, זָכִיתִי כְּבָר לִרְאוֹת בְּרָכָה בַּעֲמָלִי וִיגִיעִי, שֶׁכְּבָר נִתְפַּרְסֵם שֵׁם רַבֵּנוּ זַ"ל בְּכָל הָעוֹלָם כֻּלּוֹ, עַד שֶׁבָּאִים כְּבָר עַל רֹאשׁ הַשָּׁנָה קָרוֹב לִשְׁנֵי רִבְבוֹת אֲנָשִׁים, כֵּן יִרְבּוּ, בְּוַדַּאי עֲדַיִן נֶעֱלָם וְנִסְתָּר מֵרֹב הָאֲנָשִׁים מִי עוֹמֵד תַּחַת הַפִּרְסוּם וְהַגְדָּלַת שֵׁם הַצַּדִּיק בָּעוֹלָם, אַךְ מָה אִכְפַּת לִי, אֲנִי אֶת שֶׁלִּי אֶעֱשֶׂה, כָּל זְמַן שֶׁתִּתֵּן לִי, אָבִי שֶׁבַּשָּׁמַיִם, יָמִים וְשָׁנִים בְּזֶה הָעוֹלָם אַמְשִׁיךְ לַעֲסֹק בָּעֲבוֹדָה הַקְּדוֹשָׁה הַזּוֹ, לְקָרֵב אֶת נִשְׁמוֹת יִשְׂרָאֵל אֵלֶיךָ, עַל־ פִּי דַּעַת הַנַּחַל נוֹבֵעַ מְקוֹר חָכְמָה, עִיר וְקַדִּישׁ מִן שְׁמַיָּא נָחִית לְגָאֳלֵנוּ וּלְהוֹצִיאֵנוּ מֵהַשְּׁאוֹל תַּחְתִּית, וּלְהָאִיר עָלֵינוּ אוֹרְךָ הַגָּנוּז וְהַצָּפוּן. וְהִנֵּה זֶה זְמַן זְמַנִּים שֶׁיֵּשׁ בִּרְצוֹנִי לִרְשֹׁם עַל פְּנֵי הַכְּתָב עַל כּ־ָל־ פָּנִים רָאשֵׁי פְּרָקִים מִתּוֹלְדוֹת יְמֵי חַיָּי בְּזֶה הָעוֹלָם הָעוֹבֵר, וְאַף שֶׁאַחֲרֵי אַרְבָּעִים וּשְׁמוֹנֶה שָׁנִים - רַבּוֹת עָבְרוּ עָלַי, "כָּל מִשְׁבָּרֶיךָ וְגַלֶּיךָ עָלַי עָבָרוּ", מַחֲלֹקֶת, יִסּוּרִים, מְרִירוּת, צָרוֹת וְהַרְפַּתְקָאוֹת בְּלִי סוֹף, אֲשֶׁר בְּוַדַּאי הָרֹב שָׁכַחְתִּי, עִם כָּל זֹאת, לְמַעַן יַעַמְדוּ יָמִים רַבִּים זִכְרוֹנוֹתַי, אָמַרְתִּי אָרוּצָה מְעַט דְּבָרַי אֵלּוּ, וְכָל מִי שֶׁיִּזְכּרֹ מַה שֶּׁסִּפַּרְתִּי אוֹ מַה שֶּׁעָבַר עָלַי, יָכוֹל לְהוֹסִיף לְמַטָּה, כִּי, תְּהִלָּה לָאֵל, הֵקַמְתִּי תַּלְמִידִים לַאֲלָפִים, וְכֵן זָכִיתִי לְחַבֵּר הֲמוֹן סְפָרִים, שֶׁנִּתְפַּשְּׁטוּ בָּעוֹלָם בְּאֹפֶן מְיֻחָד יוֹצֵא מִן הַכְּלָל, אֲשֶׁר זְעִירוּן שֶׁזָּכוּ שֶׁבַּחַיִּים חַיּוּתָם יֻדְפְּסוּ סִפְרֵיהֶם בְּכָל כּ־ָךְ הַרְבֵּה מַהֲדוּרוֹת וְרִבְבוֹת טְפָסִים בְּלִי שׁוּם גֻּזְמָא כְּלָל, וְיִתְפַּשְּׁטוּ וְיָפוּצוּ בְּכָל חֶלְקֵי תֵּבֵל מַמָּשׁ, זֶה חַסְדְּךָ אֲשֶׁר עָשִׂיתָ עִמָּדִי, וּבְוַדַּאי יִרְצוּ כָּל אֵלּוּ שֶׁמְּעַיְּנִים וְלוֹמְדִים סְפָרַי לָדַעַת אֶת כָּל תּוֹלְדוֹת יְמֵי חַיָּי עַל פְּנֵי הָאֲדָמָה - מִמַּה שֶּׁעָבַר עָלַי.

וְהִנֵּה יָדוּעַ, שֶׁרַבֵּנוּ זַ"ל אָמַר עַל עַצְמוֹ, שֶׁהוּא חַי חַיִּים חֲדָשִׁים בְּכָל פַּעַם, וּבְפֵרוּשׁ גִּלָּה לָנוּ, אֲשֶׁר כָּל נְשִׁימָה, יֶעדֶע'ס דּוּךְ, הוּא אֶצְלוֹ חָדָשׁ, וְזָכָה לְחַיִּים נִצְחִיִּים, וְלָכֵן קָרָא מוֹהַרְנַ"תְּ זַ"ל אֶת שֵׁם תּוֹלְדוֹת רַבֵּנוּ זַ"ל "חַיֵּי־ מוֹהֲרַ"ן", כִּי אָמַר, שֶׁרַבֵּנוּ זַ"ל חַי בְּכָל פַּעַם חַיִּים חֲדָשִׁים, וְלָכֵן לוֹ נָאֶה שֶׁיִּקְרָא אֶת שֵׁם סֵפֶר תּוֹלְדוֹתָיו בְּשֵׁם "חַיֵּי־ מוֹהֲרַ"ן" דַּיְקָא, אֲבָל אֲנִי, סִיֵּם מוֹהַרְנַ"תְּ זַ"ל לֹא זָכִיתִי לָזֶה, רַק בְּכֹחוֹ שֶׁל רַבֵּנוּ זַ"ל הָיוּ לִי יָמִים טוֹבִים, כִּי שָׁמַרְתִּי מְאֹד עַל הַיּוֹם, וְלָכֵן קָרָא אֶת שֵׁם תּוֹלְדוֹתָיו בְּשֵׁם "יְמֵי נָתָן", כַּאֲשֶׁר נִדְפַּס הַיּוֹם: "יְמֵי מוֹהַרְנַ"תְּ" - תּוֹלְדוֹתָיו וְקוֹרוֹת חַיָּיו עַל פְּנֵי אֲדָמָה. וְהַרְבֵּה חָשַׁבְתִּי, שֶׁיַּעַן שֶׁזִּכָּנִי הַקָּדוֹשׁ בּ־ָרוּךְ־ הוּא לְגַלּוֹת וּלְפַרְסֵם אֶת דַּעַת רַבֵּנוּ זַ"ל בָּעוֹלָם, וְלַעֲשׂוֹת אֶת לִמּוּדוֹ הַטּוֹב דֶּרֶךְ כְּבוּשָׁה לָרַבִּים, וְרַבּוֹת מָסַרְתִּי אֶת נַפְשִׁי וְאֶת שְׁמִי עַל זֶה, וְעָבַר וְעוֹבֵר עָלַי מְרִירוּת וּמַחֲלוֹקוֹת וּרְדִיפוֹת מֵחֲמַת זֶה מְאֹד, עַד שֶׁלֹּא הָיָה לִי עוֹד בְּזֶה הָעוֹלָם יוֹם אֶחָד יוֹם טוֹב, עִם כָּל זֹאת בְּכֹחוֹ שֶׁל רַבֵּנוּ זַ"ל זִכָּנִי לִרְגָעִים טוֹבִים וּנְעִימִים בִּדְבֵקוּת הַבּוֹרֵא יִתְבָּרַךְ שְׁמוֹ, וְלָכֵן אָמַרְתִּי לִקְרֹא אֶת תּוֹלְדוֹת יְמֵי חַיָּי בְּשֵׁם "רִגְעֵי מוֹהֲרָא"שׁ", כִּי רְגָעִים טוֹבִים כֵּן הָיוּ לִי, וְכֵן זָכִיתִי. וְהִנֵּה בְּכָל פַּעַם עָלוּ לִי מַחֲשָׁבוֹת - מָתַי אֶקַּח אֶת עִתִּי וְעֵטִי בְּיָדִי, וְאֶרְשֹׁם הַכּלֹ עַל פְּנֵי הַכְּתָב, וְלֹא עָלְתָה בְּיָדִי מֵרֹב טִרְדּוֹתַי מִדֵּי יוֹם בְּיוֹמוֹ בְּטִרְדוֹת וְטִרְחוֹת הַכְּלָל, עַד שֶׁהַיּוֹם הַזֶּה יוֹם ד' לְסֵדֶר 'זֹאת חֻקַּת הַתּוֹרָה', ד' תַּמּוּז ה'תשס"ד, וַאֲנִי נִמְצָא בִּמְקוֹם נֹפֶשׁ בְּעִיר סֶעמְרִינְג שֶׁהוּא בִּמְדִינַת אוֹסְטְרִיָּה, קֹדֶם אוֹר הַיּוֹם, חָלַמְתִּי שֶׁקָּרְאוּ אוֹתִי לַעֲלוֹת לְסֵפֶר תּוֹרָה, וְהָיְתָה פָּרָשַׁת בְּשַׁלַּח, וְקָרְאוּ לִי עֲלִיַּת רְבִיעִי, וּבֵרַכְתִּי בִּרְכַּת הַתּוֹרָה, וְהַבַּעַל קוֹרֵא הִתְחִיל לִקְרוֹת, וּכְשֶׁבָּא לְהַמְשִׁיךְ אֶת הַקְּרִיאָה, וְהֻצְרַךְ לִגְללֹ אֶת הַסֵּפֶר תּוֹרָה, וְהִנֵּה נִפְסַק, אֵין עוֹד עַמּוּדִים, כִּי שָׁם נִסְתַּיֵּם, וְהִתְחַלְתִּי לְהַקְפִּיד אֵיךְ הִכְנִיסוּ סֵפֶר תּוֹרָה שֶׁעֲדַיִן לֹא נִשְׁלַם, אֵיךְ זֶה יָכוֹל לִהְיוֹת, שֶׁבְּדִיּוּק בְּאֶמְצַע הַקְּרִיאָה נִגְמַר. וְהָיִיתִי בְּכַעַס וּבְרֹגֶז גָּדוֹל, וְסִיַּמְתִּי אֶת הַבְּרָכָה, כְּדֵי לְהוֹצִיא סֵפֶר תּוֹרָה חָדָשׁ, וּלְהַתְחִיל לִקְרֹא מֵהַתְחָלָה, אֲבָל קֹדֶם הִתְחַלְתִּי לְגוֹלֵל אֶת הַסֵּפֶר תּוֹרָה חֲזָרָה, לִרְאוֹת מַה כָּתוּב קֹדֶם: "וַיְהִי בְּשַׁלַּח פַּרְעֹה", וְהָיוּ אוֹתִיּוֹת הַסֵּפֶר תּוֹרָה יָפוֹת מְאֹד, אֲבָל אֲנִי הָיִיתִי בְּכַעַס וְרֹגֶז וּבִקְפִידָה גְּדוֹלָה עַל זֶה, וְאֵיךְ שֶׁהִתְחַלְתִּי לְגוֹלֵל חֲזָרָה לְהַתְחָלַת הַפָּרָשָׁה. וְהִנֵּה עַל עַמּוּד אֶחָד חָלָק הָיוּ שְׁתֵּי תֵּבוֹת כְּתוּבוֹת בִּדְיוֹ: אָדָם חַוָּה, וּמִתּוֹךְ הַכְּתָב הֵם צוֹעֲקִים עָלַי: מָה אַתָּה כָּל כּ־ָךְ בְּכַעַס וּקְפֵדוּת, הֲלֹא כְּתִיב 'זֶה סֵפֶר תּוֹלְדוֹת אָדָם', אֵינְךָ יוֹדֵעַ, שֶׁכָּל מַה שֶּׁעוֹבֵר עָלֶיךָ כְּבָר נִכְתַּב אֶצְלִי, קוּם מַהֵר, וְתַתְחִיל לִכְתּבֹ אֶת כָּל הַקּוֹרוֹת אוֹתְךָ. וָאָקִיץ בְּבֶהָלָה.

וְנִסְכַּם בְּדַעְתִּי לְהַתְחִיל לִרְשֹׁם כָּל מַה שֶּׁאֲנִי זוֹכֵר, וְאַף שֶׁשָּׁכַחְתִּי רֹב הַדְּבָרִים, עִם כָּל זֹאת אֶשְׁתַּדֵּל לִרְשֹׁם כַּמָּה שֶׁאֲנִי רַק זוֹכֵר, וַאֲבַקֵּשׁ מִכָּל מִי שֶׁזּוֹכֵר מַה שֶּׁסִּפַּרְתִּי בְּמֶשֶׁךְ אַרְבָּעִים שָׁנָה, שֶׁאֲנִי עוֹסֵק לְדַבֵּר וּלְסַפֵּר וְלִדְרֹשׁ לִפְנֵי תַּלְמִידִים, בְּוַדַּאי הִזְכַּרְתִּי בְּמֶשֶׁךְ הַזְּמַן סִפּוּרִים מִימֵי נְעוּרַי וְהִתְקָרְבוּתִי, וְכֵן כְּבָר רָשׁוּם בְּהַרְבֵּה מְקוֹמוֹת אֲחֵרִים, אֲקַוֶּה שֶׁיַּאַסְפוּ אֶת זֶה, וְיִרְשְׁמוּ אֶת כָּל הַהוֹסָפוֹת לְמַטָּה, כָּל סִפּוּר לְמָקוֹם שֶׁשַּׁיָּךְ, לְמַעַן יַעֲמֹד לְיָמִים רַבִּים עַל פְּנֵי הָאֲדָמָה.

כִּי אֲנִי בָּטוּחַ, אֲשֶׁר סְפָרַי הָרַבִּים יֻדְפְּסוּ וְיַחְזְרוּ וְיֻדְפְּסוּ, וְיָפוּצוּ וְיִתְפַּשְּׁטוּ בְּיֶתֶר שְׂאֵת וּבְיֶתֶר עֹז פְּעָמִים אֵין מִסְפָּר. וְכָאן הַמָּקוֹם לְגַלּוֹת דַּעְתִּי, שֶׁכָּל סְפָרַי וְחִבּוּרַי הוּא בַּעֲבוּר כְּלַל נִשְׁמוֹת יִשְׂרָאֵל, וְאֵין עַל זֶה זְכוּת יוֹצְרִים, וְאֵין דָּבָר כָּזֶה עֲלֵיהֶם 'כָּל הַזְּכֻיּוֹת שְׁמוּרוֹת', כִּי כָּל מִי שֶׁיַּדְפִּיסֵם, תָּבוֹא עָלָיו בְּרָכָה מִן הַשָּׁמַיִם, וְאֶהְיֶה לוֹ לְמֵלִיץ יֹשֶׁר בַּשָּׁמַיִם, כִּי רַבּוֹת יָגַעְתִּי וְטָרַחְתִּי, וְרַבּוֹת סָבַלְתִּי, אֲשֶׁר רַק הַקָּדוֹשׁ בּ־ָרוּךְ־ הוּא בְּעַצְמוֹ יוֹדֵעַ, כִּי אַף בִּרְיָה לֹא תִּרְצֶה לְהַאֲמִין אֵיךְ בָּשָׂר וָדָם הָיָה יָכוֹל לִסְבּלֹ סֵבֶל כָּזֶה, וְאֵיךְ הָיָה יְלוּד אִשָּׁה בִּיכֹלֶת לַעֲברֹ בִּימֵי חַיֵּי הֶבְלוֹ מַחֲלֹקֶת וּבִזְיוֹנוֹת כָּאֵלּוּ, אַךְ בְּכֹחוֹ שֶׁל רַבֵּנוּ זַ"ל, שֶׁאֲנִי מַרְגִּישׁ שֶׁהוּא נִמְצָא תָּמִיד עִמִּי, עָבַרְתִּי עַל הַכֹּל, וּמִי שֶׁאָמַר לְעוֹלָמוֹ דַּי, יֹאמַר לְצָרוֹתַי וְסִבְלִי דַּי, וְאֶזְכֶּה לְהִכָּלֵל כְּבָר בָּאֵין סוֹף בָּרוּךְ הוּא, וְלִחְיוֹת אֶת שְׁאֵרִית יָמַי בְּנַחַת, עַד בִּיאַת גּוֹאֵל צֶדֶק, וִיקֻיַּם בִּי (בְּרֵאשִׁית ו, ח): "וְנֹחַ מָצָא חֵן בְּעֵינֵי הֲוָיָ"ה", "דִּבְרֵי חֲכָמִים בְּנַחַת יִשָּׁמְעוּ" (קֹהֶלֶת ט, יז), מֵעַתָּה וְעַד עוֹלָם.
