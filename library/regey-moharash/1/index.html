סעיף א

לֵידָתוֹ, רֶקַע הַתְּקוּפָה, גַּדְלוּת אָבִיו שֶׁלָּמַד תּוֹרָה בַּעֲנִיּוּת מֻפְלֶגֶת, עֹצֶם עִיּוּנוֹ וּפִלְפּוּלוֹ, מוּסָר לְאַהֲבַת הַתּוֹרָה, יְמֵי יַלְדּוּתוֹ הַמֻּקְדָמִים, הַמְּלַמְּדִים בַּתַּלְמוּד תּוֹרָה וּבִפְרָט ר' יַעֲקבֹ קַלְמֶנוֹבִיץ'

תוכן

נוֹלַדְתִּי בְּיוֹם רְבִיעִי כ"א אִיָּר ה'ת"ש בְּחָמֵשׁ בַּבֹּקֶר בְּתֵל-אָבִיב א לְאִמִּי הַצַּדֶּקֶת מָרַת מַלְכָּה, עָלֶיהָ הַשָּׁלוֹםב, וּלְאָבִי הַצַּדִּיק הַקָּדוֹשׁ, גָּאוֹן מֻפְלָא וּמֻפְלָג, שֶׁלָּמַד כָּל יָמָיו תּוֹרָה בְּהַתְמָדָה רַבָּה מִתּוֹךְ עֹנִי וְדֹחַקג. הוּא מָסַר אֶת נַפְשׁוֹ בִּמְסִירוּת נֶפֶשׁ גְּדוֹלָה מְאֹד עַל לִמּוּד הַתּוֹרָה הַקְּדוֹשָׁה, וְהָיָה לַמְדָּן נִפְלָא, וְרִתְּחָא דְּאוֹרַיְתָא רָתַח בּוֹ תָּמִיד, וּפִלְפֵּל אֶת עַצְמוֹ תָּמִיד עִם לוֹמְדִים עֲצוּמִיםד. הַקָּדוֹשׁ-בָּרוּךְ-הוּא חָנַן אוֹתִי עִם זִכָּרוֹן, שֶׁאֲנִי זוֹכֵר עוֹד כְּשֶׁהָיִיתִי עִם הַשְּׂעָרוֹת, וְהִכְנִיסוּ אוֹתִי לַגָּן, שֶׁהָיָה לוֹ לְר' שַׁבְּתַי (הַשַּׁמָּשׁ מֵהַשְּׁטִיבְּלָאךְ בְּמֵאָה שְׁעָרִים), וַאֲנִי זוֹכֵר אֶת הַחֲלָקֶ'ה שֶׁעָשׂוּ לִי בְּלַ"ג בָּעֹמֶר, וְהָיְתָה שִׂמְחָה גְּדוֹלָה מְאֹד, וִיהוּדִי זָקֵן רָקַד עִמִּי עַל כְּתֵפוֹ, וְאַחַר-כָּךְ הִכְנִיסוּ אוֹתִי לַחֵדֶר, שֶׁהָיָה בַּשְׁטִיבְּלֶעךְ בִּשְׁכוּנַת בֵּית יִשְׂרָאֵל. הַמְּלַמֵּד שֶׁלִּי הָיָה ר' דָּוִד זַלְצְמַן, וַאֲנִי זוֹכֵר אֵיךְ שֶׁלָּמַד אִתִּי אוֹתִיּוֹת אָלֶ"ף-בֵּי"ת, עַד שֶׁבְּסוֹף הַשָּׁנָה כְּבָר יָכֹלְתִּי לִקְרֹא בַּסִּדּוּר לְבַדה, וְאַחַר-כָּךְ נִכְנַסְתִּי בְּכִתָּה ב' אֵצֶל ר' חַיִּים זֶעלִיג, וְלָמַדְתִּי אֶצְלוֹ שָׁנָה חֻמָּשׁ, אַחַר-כָּךְ נִכְנַסְתִּי בְּכִתָּה ג' אֵצֶל ר' יַעֲקבֹ קַלְמְנוֹבִיץ (שֶׁהָיָה מְפֻרְסָם בְּשֵׁם רַבִּי יַענְקְל מְלַמֵּדו), וְהוּא כְּבָר לָמַד מִלְּבַד חֻמָּשׁ עִם הַיְלָדִים נַ"ךְ (אֶצְלוֹ עֲדַיִן לֹא לָמְדוּ רַשִּׁ"י), וְהִסְפִּיק בְּשָׁנָה אַחַת לִגְמרֹ יְהוֹשֻׁעַ, שׁוֹפְטִים, שְׁמוּאֵל א', וּשְׁמוּאֵל ב', עַד מְלָכִים א' פֶּרֶק ח', וְכֹחַ הַהֶסְבֵּר שֶׁלּוֹ הָיָה לְהַפְלִיא, וְהָיָה מַמְחִישׁ אֶת הַלִּמּוּד בְּצוּרָה כָּזוֹ, שֶׁרָאוּ כָּל מַה שֶּׁלָּמַד אֶת זֶה בְּחוּשׁ מַמָּשׁ חַי, אֶת יְהוֹשֻׁעַ אֵיךְ שֶׁעוֹבֵר אֶת הַיַּרְדֵּן, וְאֵיךְ שֶׁלּוֹחֵם עִם הַל"א מְלָכִים וְכוּ', וְכוּ', הִמְחִישׁ אֶת כָּל פְּרָט וּפְרָט בְּצוּרָה כָּזוֹ שֶׁהוּא מַמָּשׁ חַי, וְכֵן כָּל מַה שֶּׁעָבַר עַל כָּל שׁוֹפֵט וְשׁוֹפֵט, וְכֵן מַה שֶּׁעָבַר עַל חַנָּה וְעֵלִי הַכֹּהֵן עַד שֶׁנּוֹלַד שְׁמוּאֵל, וְכָל מַה שֶּׁעָבַר עָלָיו, וְכֵן מַה שֶּׁקָּרָה לְשָׁאוּל, לְדָוִד, הַכּלֹ הָיָה חַי, וְעַד הַיּוֹם שֶׁזֶּה כְּבָר כִּמְעַט שִׁשִּׁים שָׁנָה, אֲנִי זוֹכֵר אֶת זֶה כְּמוֹ שֶׁרַק עַכְשָׁו לָמַדְתִּי אֶת זֶה, וַאֲנִי נִמְצָא בְּאוֹתָהּ תְּקוּפָה שֶׁל יְהוֹשֻׁעַ בִּן נוּן וְהַשּׁוֹפְטִים וּשְׁמוּאֵל הַנָּבִיא וְשָׁאוּל הַמֶּלֶךְ וְדָוִד הַמֶּלֶךְ, כִּי הִמְחִישׁ אֶת הַלִּמּוּד לַתַּלְמִידִים בְּצוּרָה יוֹצֵאת מִן הַכְּלָל, עַד שֶׁכֻּלָּם הֵבִינוּ הֵיטֵב אֶת הַלִּמּוּד בְּאֹפֶן נַעֲלֶה מְאֹד.

עִם כָּל זֹאת אָבִי מוֹרִי לָמַד תּוֹרָה בְּשִׂמְחָה עֲצוּמָה, וּפִלְפֵּל עִם גְּדוֹלֵי הַלּוֹמְדִים, כְּאִלּוּ לֹא קָרָה שׁוּם דָּבָר, וְהֶחֱיָה אֶת עַצְמוֹ עִם הַתּוֹרָה, וְכֵן בְּכָל פַּעַם הָיָה נֵס אַחֵר שֶׁיִּהְיֶה עַל צָרְכֵי שַׁבָּת, עִם כָּל זֹאת הִתְפָּאֵר תָּמִיד אָבִי, שֶׁאֵלּוּ הַשָּׁנִים הָיוּ לוֹ הַשָּׁנִים הֲכִי יָפוֹת בְּחַיָּיו, כִּי הָיָה יָכוֹל לִלְמֹד תּוֹרָה מִתּוֹךְ הַדְּחָק, שֶׁאָז הַלֵּב שָׁבוּר וְהַמֹּחַ דָּבוּק בַּתּוֹרָה, אֲשֶׁר אֵין טוֹב מִזֶּה, גַּם בַּהַתְחָלָה לֹא הָיוּ לָהֶם מִטּוֹת לִישֹׁן, וְיָשְׁנוּ עַל הָרִצְפָּה, שֶׁשָּׂמוּ סָדִין קָרוּעַ, וְעַל זֶה שָׁכְבוּ, וְעִם עוֹד סָדִין קָרוּעַ הִתְכַּסּוּ, וְהָיָה תָּמִיד אָבִי מוֹרִי מִתְפָּאֵר, שֶׁקִּיֵּם בִּשְׁלֵמוּת אֶת מַאֲמָרָם זַ"ל (אָבוֹת, פֶּרֶק ו'): כָּךְ דַּרְכָּהּ שֶׁל תּוֹרָה - פַּת בַּמֶּלַח תּאֹכֵל, וּמַיִם בַּמְּשׂוּרָה תִּשְׁתֶּה, וְעַל הָאָרֶץ תִּישַׁן, אִם אַתָּה עוֹשֶׂה כֵן, אַשְׁרֶיךָ בָּעוֹלָם הַזֶּה וְטוֹב לָךְ לָעוֹלָם הַבָּא, וְהָיָה מְדַיֵּק מַה הַלָּשׁוֹן: "אִם אַתָּה עוֹשֶׂה כֵן", הֲרֵי כְּבָר אָמַר: כָּךְ דַּרְכָּהּ שֶׁל תּוֹרָה? אֶלָּא לֹא לִדְרָשָׁא קָאָתֵי הַתַּנָּא, שֶׁכָּךְ דַּרְכָּהּ שֶׁל תּוֹרָה, אֶלָּא "אִם אַתָּה עוֹשֶׂה כֵן" בְּפֹעַל מַמָּשׁ, אָז אַתָּה מַרְגִּישׁ נְעִימוּת יְדִידוּת הַתּוֹרָה, כִּי מֵאַחַר שֶׁאֵין לְךָ אֶלָּא פַּת בְּצִמְצוּם, וּמַיִם כּוֹס תֵּה, וְעַל הָאָרֶץ תִּשְׁכַּב, פְּשׁוּטוֹ כְּמַשְׁמָעוֹ, אָז דַּיְקָא אַשְׁרֶיךָ בָּעוֹלָם הַזֶּה וְטוֹב לָךְ לָעוֹלָם הַבָּא.

הערות

  1. אבִּשְׁנַת ת"ש כְּבָר הָיוּ הַגְּזֵרוֹת בְּהוּנְגַרְיָה עַל הַיְּהוּדִים, וּמִי שֶׁלֹּא הָיְתָה לוֹ אֶזְרָחוּת הוּנְגָרִית, שָׁלְחוּ אוֹתוֹ אֶל הַגְּבוּל, שֶׁמִּצַּד אֶחָד הָיוּ הָרוֹמָנִים, יִמַּח שְׁמָם, וּמִצַּד שֵׁנִי הָיוּ הַהוּנְגָרִים, יִמַּח שְׁמָם, וְכָכָה גֵּרְשׁוּ אֶת הַיְּהוּדִים אֵלּוּ מוּל אֵלּוּ, וְהַהוּנְגָרִים יָרוּ עֲלֵיהֶם מִצַּד אֶחָד, וְהָרוֹמָנִים יָרוּ עֲלֵיהֶם מִצַּד שֵׁנִי, וְכָכָה נֶהֶרְגוּ הֲמוֹן יְהוּדִים שֶׁלֹּא הָיְתָה לָהֶם שׁוּם אֶזְרָחוּת, וְיַעַן שֶׁלְּאָבִי לֹא הָיְתָה שׁוּם אֶזְרָחוּת, וְלֹא נְיָרוֹת וְכוּ', וְלָכֵן כְּשֶׁהָיָה הָרַב דִּקְהַל חֲסִידִים בָּעִיר טָאקַאי, מָסְרוּ אוֹתוֹ שָׁם בִּשְׁנַת תרצ"ט, שֶׁאֵין לוֹ נְיָרוֹת, אֲזַי הֻצְרַךְ לִבְרֹחַ מִשָּׁם עַד עִיר הַבִּירָה בּוּדַאפֶּעסְט, וְהָיָה טָמִיר וְנֶעֱלָם שָׁם שָׁנָה אַחַת. פַּעַם הָלַךְ בָּרְחוֹב, וְתָפְסוּ אוֹתוֹ הַחַיָּלִים הַהוּנְגָרִים, יִמַּח שְׁמָם, וְהֵבִיאוּ אוֹתוֹ לְמַעֲצָר אֶל הַמְפַקֵּד שֶׁל תַּחֲנַת הַמִּשְׁטָרָה, וְהוּא הָיָה רָשָׁע מְרֻשָּׁע, אִישׁ קָשׁוּחַ מְאֹד, וְצָעַק מוּל אָבִי: מָתַי בָּאתֶם לְבוּדַאפֶּעסְט? עָנָה וְאָמַר לוֹ אָבִי בְּבִטָּחוֹן עַצְמִי חָזָק מְאֹד: אֶתְמוֹל. וְשָׁאַל אוֹתוֹ הַמְּפַקֵּד: וְאֵיפֹה הַנְּיָרוֹת שֶׁלְּךָ? עָנָה וְאָמַר לוֹ אָבִי: הִשְׁאַרְתִּי אוֹתָם בַּבַּיִת (אֲשֶׁר בֶּאֱמֶת לֹא הָיָה לוֹ דָּבָר כָּזֶה). אָז צָעַק הַמְּפַקֵּד בִּרְצִיחָה: אֲנִי רוֹצֶה שֶׁעַד עֲשָׂרָה יָמִים תָּבוֹא חֲזָרָה עִם הַנְּיָרוֹת, וְעַכְשָׁו לֵךְ (וְזֶה הָיָה נֵס בְּתוֹךְ נֵס, כִּי כְּשֶׁנּוֹפְלִים בִּידֵי הָרְשָׁעִים, כְּבָר לֹא יְכוֹלִים לָצֵאת, וּבִפְרָט לִתְקוּפָה אֲרֻכָּה שֶׁל עֲשָׂרָה יָמִים). אָז אָמַר לוֹ אָבִי בְּבִטָּחוֹן עַצְמִי חָזָק: אֲדוֹנִי הַמְּפַקֵּד, הִנֵּה אֲנִי רַק אֵצֵא, וְעוֹד פַּעַם יִתְפְּסוּ אוֹתִי, תֵּן לִי אֶת זֶה בִּכְתָב, שֶׁעַד עֲשָׂרָה יָמִים אֲנִי צָרִיךְ לַחֲזֹר. וּפֶלֶא עַל גַּבֵּי פֶּלֶא, שֶׁנָּתַן לוֹ בִּכְתָב, שֶׁבַּיּוֹם הָעֲשִׂירִי הוּא מֻכְרָח לַחֲזֹר לְתַחֲנַת הַמִּשְׁטָרָה עִם הַנְּיָרוֹת שֶׁלּוֹ, וּבְתוֹךְ עֲשָׂרָה יָמִים הָאֵלּוּ, אָבִי הִזְדָּרֵז מְאֹד, וְקָרָא אֶת אִמִּי וְאֶת אָחִי, וְאָז הָיְתָה אֳנִיָּה שֶׁל הַמַּעְפִּילִים מוּכָנָה לִנְסֹעַ לְאֶרֶץ יִשְׂרָאֵל, וְאוֹמֵר וְעוֹשֶׂה, הֶחְלִיט שֶׁהוּא נוֹסֵעַ לְאֶרֶץ יִשְׂרָאֵל, אֲבָל בִּקְּשׁוּ מִמֶּנּוּ סְכוּם עֲנָקִי שֶׁלֹּא הָיְתָה לוֹ אֲפִלּוּ אֲגוֹרָה אַחַת, וְנַעֲשָׂה לוֹ נֵס שֶׁקְּהִלַּת טָאקַאי שָׁלְחוּ לוֹ שְׁלִישׁ מֵהַכֶּסֶף, וְכֵן הַגָּאוֹן הַצַּדִּיק רַבִּי יוֹנָתָן שְׁטַייף, זֵכֶר צַדִּיק לִבְרָכָה, הָרַב מִבּוּדַאפֶּעסְט הִשִּׂיג בִּשְׁבִילוֹ שְׁלִישׁ מֵהַכֶּסֶף, וְעוֹד יְהוּדִי בְּשֵׁם ר' חַיִּים שְׁטֶרְן זַ"ל, שֶׁאֶצְלוֹ הִתְחַבֵּא אָבִי כָּל הַשָּׁנָה, נָתַן אֶת שְׁלִישׁ הַכֶּסֶף, וְכָךְ בְּתוֹךְ יוֹמַיִם שְׁלֹשָׁה גִּיֵּס אֶת כָּל הַכֶּסֶף, וּכְבָר שִׁלֵּם, וְנִכְנְסוּ בָּאֳנִיָּה, וְנָסְעוּ כָּךְ חֹדֶשׁ יָמִים (כִּי הָאֳנִיָּה הָיְתָה יְשָׁנָה לְמַעְלָה מִמֵּאָה שָׁנָה, מֵאָז שֶׁבָּנוּ אֶת זֶה, וְזֶה הָיָה לְכַתְּחִלָּה לְהוֹבִיל בְּהֵמוֹת, וְעַכְשָׁו נִדְחֲקוּ שָׁם שְׁבַע-מֵאוֹת יְהוּדִים. וְלָכֵן הִפְלִיגָה מְאֹד מְאֹד לְאַט, וּמַה שֶּׁעָבַר עֲלֵיהֶם בַּנְּסִיעָה הַזּוֹ, יִקְצְרוּ הֲמוֹן יְרִיעוֹת, אֵילוּ נִסִּים נִגְלִים הָיוּ לָהֶם, כִּי הָאַנְגְלִים, יִמַּח שְׁמָם, הִטְבִּיעוּ אֳנִיּוֹת עַל הַדֶּרֶךְ שֶׁנָּסְעָה לְאֶרֶץ יִשְׂרָאֵל, וְלִפְנֵי הָאֳנִיָּה שֶׁלָּהֶם הִטְבִּיעוּ אֳנִיָּה מְלֵאָה עִם יְהוּדִים, וְכֵן אַחֲרֵי הָאֳנִיָּה שֶׁלָּהֶם הִטְבִּיעוּ אֳנִיָּה מְלֵאָה עִם יְהוּדִים, הי"ד, וּבְנֵס נִפְלָא לֹא הִבְחִינוּ בָּאֳנִיָּה הַזּוֹ), עַד שֶׁהִגִּיעוּ לְחוֹף תֵּל-אָבִיב, וְאָז עָצְרוּ אֶת כֻּלָּם הָאַנְגְלִים שֶׁשָּׁלְטוּ אָז בְּאֶרֶץ יִשְׂרָאֵל, וְהִכְנִיסוּ אוֹתָם בְּמַחֲנֶה צְבָאִי בְּעַתְלִית (אֲשֶׁר זֶה בְּעַצְמוֹ הָיָה נֵס, כִּי רֹב הַיְּהוּדִים שֶׁהִגִּיעוּ בָּאֳנִיּוֹת מַעְפִּילִים, הִכְרִיחוּ אוֹתָם לַחֲזֹר לְקַפְרִיסִין), וְשָׁהוּ שָׁם, וְהִנֵּה יְהוּדִים רַחְמָנִים בְּנֵי רַחְמָנִים בָּאוּ אֶל הַמַּחֲנֶה הַצְּבָאִי, וְרָצוּ לְהוֹצִיא אֶת הַיְּהוּדִים שֶׁהֵם קְרוֹבֵי מִשְׁפַּחְתָּם, וְלָכֵן כָּל אֶחָד נִגַּשׁ אֶל אַחֵר, וְאָמַר לוֹ: אָחִי, אֵיךְ אַתָּה מַרְגִּישׁ?! וְזֶה אָמַר אֶל אַחֵר: גִּיסִי, אֵיךְ אַתָּה מַרְגִּישׁ?! וְכָךְ כָּל אֶחָד עָשָׂה אֶת עַצְמוֹ קָרוֹב מִשְׁפָּחָה לְאֵלּוּ הָאֲסִירִים, וְכָךְ שִׁחְרְרוּ אוֹתָם הָאַנְגְלִים, וִיהוּדִי יָקָר בְּשֵׁם ר' יוֹסֵף בִּילֶער זַ"ל, הוּא נִגַּשׁ אֶל אָבִי, וְחִבֵּק אוֹתוֹ וְנִשֵּׁק אוֹתוֹ: דּוֹדִי, אֵיךְ אַתָּה מַרְגִּישׁ?! וְכָךְ שִׁחְרְרוּ אוֹתָם (כִּי זֶה הָיָה הַחֹק אָז, שֶׁמִּי שֶׁיֵּשׁ לוֹ קְרוֹבֵי מִשְׁפָּחָה יָכוֹל לָצֵאת אֲלֵיהֶם, וּמִי שֶׁאֵין לוֹ קְרוֹבֵי מִשְׁפָּחָה, שׁוֹלְחִים אוֹתוֹ חֲזָרָה לְקַפְרִיסִין, וְאָז רָאוּ אֶת מַעֲלָתָם שֶׁל יְהוּדִים רַחְמָנִים בְּנֵי רַחְמָנִים, שֶׁבָּאוּ בַּהֲמוֹנֵיהֶם, וְכָל אֶחָד אִמֵּץ מִשְׁפָּחָה אַחֶרֶת שֶׁהֵם קְרוֹבֵי מִשְׁפַּחְתּוֹ). וְהַיְּהוּדִי הַזֶּה הִכְנִיס אֶת אָבִי וְאֶת אִמִּי וְאֶת אָחִי שֶׁהָיָה אָז בְּגִיל אַרְבַּע, אֶל תּוֹךְ בֵּיתוֹ, וְכָךְ הָיָה, שֶׁאָבִי גָּר בַּהַתְחָלָה בְּתֵל אָבִיב, כִּי הוּא בָּא בְּעֵרֹם וּבְחֹסֶר כֹּל, וְהָיָה מֻכְרָח לְהִתְגּוֹרֵר בְּבֵית הַיְּהוּדִי הַזֶּה.
  2. באִמִּי סִפְּרָה לִי כְּשֶׁהִיא הָיְתָה בְּהֵרָיוֹן אִתִּי, יוֹם אֶחָד הִיא לָקְחָה אֶת אָחִי לַחֵדֶר, וּבְאוֹתוֹ יוֹם הָיָה מְאֹד חַם, וְהִיא לֹא אָכְלָה אָז, כִּי פָּשׁוּט לֹא הָיָה לָהֶם מַה לֶּאֱכֹל, וְלָכֵן עַל הַדֶּרֶךְ חֲזָרָה הִיא הִתְעַלְּפָה וְנָפְלָה בָּרְחוֹב, וּכְשֶׁהִיא הִתְעוֹרְרָה, מָצְאָה אֶת עַצְמָהּ בְּבֵית חוֹלִים (כַּנִּרְאֶה שֶׁהֵבִיאוּ אַמְבּוּלַנְס, וְלָקְחוּ אוֹתָהּ לְבֵית-חוֹלִים), וְהָרוֹפְאִים קָבְעוּ שֶׁהַוָּלָד מֵת בְּבִטְנָהּ, וְלָכֵן עָלֶיהָ לַעֲשׂוֹת הַפָּלָה, וְיַעַן שֶׁאִמִּי הָיְתָה בִּרְיָה גְּדוֹלָה וּזְרִיזָה מְאֹד, וּכְבָר רָאֲתָה אֵיפֹה שֶׁנָּפְלָה, אָז אָמְרָה לָרוֹפְאִים: תְּנוּ לִי עוֹד שְׁעָתַיִם שָׁלֹשׁ לְהָכִין אֶת עַצְמִי לַהַפָּלָה וְהִסְכִּימוּ, וּבֵין כָּךְ בָּרְחָה מִבֵּית-הַחוֹלִים (וּתְהִלָּה לָאֵל, נוֹלַדְתִּי בָּרִיא וְשָׁלֵם, כַּנִּרְאֶה שֶׁהַסָּמֶ"ךְ-מֵ"ם כְּבָר רָצָה לְהָמִית אוֹתִי עוֹד בְּבֶטֶן אִמִּי). הַבְּרִית הָיְתָה בְּבֵית-הַכְּנֶסֶת שֶׁל הַיְּהוּדִי הַנַּ"ל (כִּי הָיָה לוֹ בֵּית-הַכְּנֶסֶת סָמוּךְ לְדִירָתוֹ), וְהַסַּנְדָק הָיָה יְהוּדִי שֶׁבָּא אֶל אָבִי, וְאָמַר שֶׁהוּא מְנַדֵּב אֶת כָּל צָרְכֵי סְעֻדַּת הַבְּרִית, אַךְ הוּא רוֹצֶה לִהְיוֹת סַנְדָּק, וְיַעַן שֶׁלֹּא הָיָה אָז כְּלוּם לְאַף אֶחָד, וְהַיְּהוּדִי הַזֶּה הֵבִיא שֶׁפַע, שֶׁהִסְפִּיק לַהֲמוֹן עֲנִיִּים לֶאֱכלֹ, הִסְכִּים אָבִי, וְהָיָה רָגִיל לוֹמַר, שֶׁעַד הַיּוֹם הוּא לֹא יוֹדֵעַ מִי הוּא זֶה הַיְּהוּדִי, כִּי לִפְנֵי הַבְּרִית לֹא רָאָה אוֹתוֹ אַף פַּעַם, וְאַחֲרֵי הַבְּרִית כְּבָר לֹא רָאָה אוֹתוֹ אַף פַּעַם, וְעַד הַיּוֹם זֶה אֶצְלוֹ פֶּלֶא מִי הָיָה הַיְּהוּדִי הַזֶּה, וְשָׁאַל וְחָקַר אֶת הָאֲנָשִׁים אִם רָאוּ פַּעַם אֶת הָאָדָם הַזֶּה, וְאַף אֶחָד לֹא יָדַע מִי הוּא, כִּי לֹא הִכִּירוּ אוֹתוֹ.
  3. גכִּי הָיוּ יָמִים שֶׁהָיָה חָסֵר לֶחֶם בַּבַּיִת, פְּשׁוּטוֹ כְּמַשְׁמָעוֹ, וְאִמִּי מוֹרָתִי מָסְרָה אֶת נַפְשָׁהּ לְהַשִּׂיג בְּדֶרֶךְ לֹא דֶּרֶךְ קֶמַח, וְאָפְתָה בְּעַצְמָהּ כַּמָּה פִּתּוֹת, (מַמָּשׁ לֶחֶם עֹנִי), וּבָזֶה הֶחְיוּ אֶת עַצְמָם עִם כּוֹס תֵּה, וְלוּלֵא שֶׁהָאִמָּא הִשִּׂיגָה בְּדֶרֶךְ לֹא דֶּרֶךְ בְּכָל פַּעַם קְצָת קֶמַח, הָיָה גּוֹוְעִין בָּרָעָב, וְאִי אֶפְשָׁר לְהַסְבִּיר אֶת זֶה לְאָדָם שֶׁלֹּא עָבַר אֶת זֶה בְּחַיָּיו, שֶׁלֹּא יִהְיֶה בַּבַּיִת אֲפִלּוּ לֶחֶם יָבֵשׁ וְכוּ'.
  4. דשֶׁהָיוּ בַּתְּקוּפָה הַהִיא בִּירוּשָׁלַיִם לוֹמְדִים מֻפְלָגִים, שֶׁהָיוּ בְּקִיאִים בְּשַׁ"ס וּפוֹסְקִים בִּבְקִיאוּת נִפְלָאָה מְאֹד, וְהָיָה שָׁגוּר עַל לְשׁוֹנָם, וְגַם הֵם הָיוּ עֲנִיִּים מְרוּדִים, וְהָיוּ צְנוּעִים מְאֹד, שֶׁהָעוֹלָם לֹא יָדְעוּ מֵהֶם כְּלָל, וְהָיָה אָבִי מוֹרִי תָּמִיד מִתְפָּאֵר, כִּי הַוִּכּוּחִים וְהַפִּלְפּוּלִים עִמָּהֶם הָיָה מַמָּשׁ תַּעֲנוּג גַּן-עֵדֶן, וְאַף שֶׁדֶּרֶךְ הַלַּמְדָנִים, שֶׁלֹּא מְוַתְּרִים אֶחָד לַשֵּׁנִי, וּכְשֶׁמִּתְוַכְּחִים, הַקּוֹלוֹת עוֹלִים מִדֵּי פַּעַם יוֹתֵר גָּבוֹהַּ, וְאֶחָד מִתְנַגֵּחַ עִם הַשֵּׁנִי, עִם כָּל זֹאת הַוִּכּוּחַ בְּלִמּוּדָם הָיָה עִם אֱמֶת, כִּי מֵאַחַר שֶׁהֵם הָיוּ עֲנִיִּים מְרוּדִים, וְלֹא הָיָה לָהֶם שׁוּם דָּבָר רַק הַתּוֹרָה, וּבְנֵי-אָדָם לֹא הִסְתַּכְּלוּ עֲלֵיהֶם לַעֲזֹר לָהֶם, אֲזַי כָּל חוּשָׁם וּמֹחָם הָיָה רַק בַּתּוֹרָה, שֶׁזֶּה נִקְרָא תּוֹרָה לִשְׁמָהּ, שֶׁלֹּא לוֹמְדִים בִּשְׁבִיל שֵׁם, אוֹ פִּרְסוּם, אוֹ כָּבוֹד, כִּי אֵיזֶה שֵׁם כְּבָר הָיָה לְלַמְּדָן עָנִי, אַף אֶחָד לֹא יֵדַע אֶת מַהוּתוֹ, וּמֶה הָיָה שַׁיָּךְ פִּרְסוּם אוֹ כָּבוֹד, בְּשָׁעָה שֶׁאֵין בַּבַּיִת אֲפִלּוּ חֲתִיכַת לֶחֶם לְהַחְיוֹת אֶת עַצְמוֹ לְיוֹם אֶחָד (וַאֲנִי זוֹכֵר, כְּשֶׁעוֹד שָׁכַבְתִּי בַּעֲרִיסַת תִּינוֹק, כְּבָר מֵאָז שָׁמַעְתִּי בַּבַּיִת קוֹלוֹת שֶׁל מְרִיבוֹת בַּתּוֹרָה, וְכָךְ גָּדַלְתִּי בְּבַיִת שֶׁתָּמִיד הִתְוַכְּחוּ בַּתּוֹרָה, וְקוֹל הַתּוֹרָה הָיָה נִשְׁמָע בֵּין כָּתְלֵי הַבַּיִת).
  5. הוַאֲנִי זוֹכֵר, שֶׁמֵּאָז שֶׁיָּכֹלְתִּי לְחַבֵּר אֶת הָאוֹתִיּוֹת בַּסִּדּוּר, לֹא הִפְסַקְתִּי מִלְּהִתְפַּלֵּל, אֲפִלּוּ כְּשֶׁהָיִיתִי בֵּינִי לְבֵין עַצְמִי, כְּשֶׁחָזַרְתִּי מֵהַחֵדֶר, תָּמִיד הָיִיתִי קוֹרֵא שַׁחֲרִית, מִנְחָה, מַעֲרִיב בְּיַחַד, בְּשִׂמְחָה עֲצוּמָה, שֶׁאֲנִי יָכוֹל לִקְרֹא כְּבָר אֶת הַסִּדּוּר, כִּי עֲדַיִן לֹא הֵבַנְתִּי מַה זֶּה זְמַנֵּי הַתְּפִלָּה, כִּי הָיִיתִי עֲדַיִן קָטָן מְאֹד, אֲבָל רָצִיתִי תָּמִיד לַחְזֹר אַחַר הַשָּׁלֹשׁ תְּפִלּוֹת, וְלָכֵן הָיִיתִי רָגִיל לוֹמַר אוֹתָם, אֲפִלּוּ לְעֵת עֶרֶב כְּשֶׁחָזַרְתִּי מֵהַחֵדֶר, הָעִקָּר שֶׁלֹּא אֶשְׁכַּח מַה שֶּׁלָּמַדְתִּי אֶת קְרִיאַת הַתֵּבוֹת שֶׁבַּסִּדּוּר, וְהָיִיתִי מָלֵא שִׂמְחָה, שֶׁאֲנִי כְּבָר יָכוֹל לִקְרֹא בַּסִּדּוּר.
  6. והוּא הָיָה אֶחָד מֵאַנְשֵׁי שְׁלוֹמֵנוּ חָסִיד בְּרֶסְלֶב, אִישׁ יָרֵא וְחָרֵד מְאֹד, וְהָיָה מְאֹד אָדוּק וְדָבוּק אֶל הָרַב הֶחָסִיד הַמְפֹאָר ר' אַבְרָהָם שְׁטֶרִינְהַארְץ זַ"ל, וְהָיָה שָׁנִים רַבּוֹת בַּעַל שַׁחֲרִית בְּרֹאשׁ הַשָּׁנָה בְּמֵרוֹן, הוּא הָיָה בַּעַל קוֹרֵא נִפְלָא מְאֹד, וַאֲנָשִׁים מְסַפְּרִים, שֶׁטַּעַם הַקְּרִיאָה שֶׁלּוֹ הָיָה מָתוֹק מִדְּבַשׁ, מִלְּבַד זֹאת הָיָה עוֹבֵד ה' גָּדוֹל מְאֹד, וְהָיָה מַרְבֶּה בְּהִתְבּוֹדְדוּת. וְסִפֵּר לִי אֶחָד מֵאַנְשֵׁי שְׁלוֹמֵנוּ, שֶׁשָּׁמַע מִמֶּנּוּ, שֶׁדַּרְכּוֹ לָקוּם כָּל יוֹם הַרְבֵּה שָׁעוֹת קֹדֶם אוֹר הַיּוֹם, וְהָיוּ לוֹ אֶת הַסְּדָרִים שֶׁלּוֹ בַּתּוֹרָה, בְּהִתְבּוֹדְדוּת, וְהָיָה לוֹ שִׁעוּר מְיֻחָד בְּהִתְבּוֹדְדוּת לְהִתְפַּלֵּל כָּל יוֹם קֹדֶם אוֹר הַיּוֹם, עַל כָּל תַּלְמִיד שֶׁלָּמַד אֶצְלוֹ, שֶׁיַּצְלִיחַ בְּלִמּוּדוֹ, וּפַעַם אָמַר, שֶׁבְּיוֹם שֶׁלֹּא מִתְפַּלֵּל קֹדֶם אוֹר הַיּוֹם עַל כָּל תַּלְמִיד וְתַלְמִיד, יֵשׁ לוֹ בְּאוֹתוֹ יוֹם אֵיזוֹ תַּקָּלָה בַּחֵדֶר, שֶׁהַיֶּלֶד לֹא מֵבִין מַה שֶׁלּוֹמְדִים. וְאֶחָד מֵאַנְשֵׁי שְׁלוֹמֵנוּ סִפֵּר לִי שֶׁשָּׁמַע פַּעַם מִמֶּנּוּ, שֶׁמַּרְגִּישׁ שֶׁזֹּאת פָּעַל אֵצֶל הַשֵּׁם יִתְבָּרַךְ, שֶׁכָּל תַּלְמִיד שֶׁלָּמַד אֶצְלוֹ, לֹא יֵצֵא לְתַרְבּוּת רָעָה (מִכָּל זֶה רוֹאִים אֵיךְ שֶׁהָיָה מָסוּר כָּל-כָּךְ לְתַלְמִידָיו, וְלָכֵן לֹא בְּחִנָּם שֶׁכָּל הַתַּלְמִידִים שֶׁלּוֹ הִצְלִיחוּ כָּל-כָּךְ בְּלִמּוּד, וְזָכְרוּ הֵיטֵב מַה שֶּׁלָּמְדוּ). וְכֵן שָׁמַעְתִּי מִכַּמָּה מֵאַנְשֵׁי שְׁלוֹמֵנוּ שֶׁהָיוּ פַּעַם בְּבֵיתוֹ, וְרָאוּ אֵיךְ שֶׁבְּהַרְבֵּה סְפָרִים הָיָה כָּתוּב בְּסוֹף הַסֵּפֶר בְּזוֹ הַלָּשׁוֹן: "סִיַּמְתִּי אֶת הַסֵּפֶר כֻּלּוֹ". וְשָׁאֲלוּ אוֹתוֹ: מַה זֶּה? עָנָה וְאָמַר: הֱיוֹת שֶׁרַבֵּנוּ זַ"ל הִזְהִירָנוּ מְאֹד וְאָמַר (שִׂיחוֹת-הָרַ"ן, סִימָן כח), שֶׁרָאוּי לְאָדָם שֶׁיֵּלֵךְ וְיַעֲברֹ בְּזֶה הָעוֹלָם בְּכָל הַסְּפָרִים הַקְּדוֹשִׁים וְלִלְמֹד כֻּלָּם, כְּדֵי שֶׁיִּהְיֶה בְּכָל מָקוֹם, אֲזַי אֲנִי מִשְׁתַּדֵּל לְקַיֵּם אֶת זֶה בִּתְמִימוּת וּפְשִׁיטוּת, וּבְאֵיזֶה סֵפֶר שֶׁזָּכִיתִי לַעֲברֹ, עַל זֶה אֲנִי רוֹשֵׁם: "סִיַּמְתִּי אֶת הַסֵּפֶר כֻּלּוֹ", כְּדֵי שֶׁאֵדַע, שֶׁאֶת הַסֵּפֶר הַזֶּה כְּבָר סִיַּמְתִּי, וְעָלַי לְהַתְחִיל סֵפֶר אַחֵר, וְהָיָה מִתְאַנֵּחַ: מִי יִתֵּן, שֶׁאֶזְכֶּה לַעֲבֹר אֶת כָּל הַסְּפָרִים שֶׁיֵּשׁ (מִכָּל זֶה רוֹאִים אֶפֶס קְצָת מִקְּצָת מִן הַמִּקְצָת גְּדֻלָּתוֹ, שֶׁמִּבַּחוּץ חָשְׁבוּ שֶׁזֶּה סְתָם יְהוּדִי מְלַמֵּד, אֲבָל הוּא הָיָה עוֹבֵד הַשֵּׁם גָּדֹל מְאֹד, וְהָיָה מַרְבֶּה בְּהִתְבּוֹדְדוּת כָּל יְמֵי חַיָּיו).
