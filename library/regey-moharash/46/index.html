סעיף מו

שְׁנַת תשכ"ה – שְׁנַת הַשִּׂיא בְּעִנְיַן הַהַדְפָּסָה

תוכן

בֵּין כָּךְ הִגִּיעַ רֹאשׁ הַשָּׁנָה תשכ"ה, וְנִתְקַבַּצְנוּ יַחַד, אֲבָל רָאִיתִי שֶׁהַרְבֵּה אֲנָשִׁים שֶׁהָיוּ לָהֶם אִינְטֶרֶסִים אִישִׁיִּים, הִתְחִילוּ לְהָסִית נֶגְדִּי, כִּי יָחִיד יִזְכֶּה לְהַדְפִּיס כָּל-כָּךְ הַרְבֵּה סְפָרִים, זֶה עֲדַיִן לֹא הָיָה, בִּפְרָט לִמְכּרֹ אֶת זֶה בִּמְחִיר סִמְלִי תַּחַת מְחִיר הַקֶּרֶן בְּלִי עֶזְרָה מִשּׁוּם בִּרְיָה, וְאַדְרַבָּה עִם הִתְנַגְּדוּת מִכָּל הַצְּדָדִים, וְהִנֵּה בִּשְׁנַת תשכ"ה אָז הָיָה הַשִּׂיא לְהַדְפָּסָה וְלַהֲפָצָה, שֶׁהֵפַצְתִּי בְּמֶרֶץ גָּדוֹל מִדֵּי יוֹם בְּיוֹמוֹ, וּמִדֵּי שָׁבוּעַ בְּשָׁבוּעַ, גֻּזְמָאוֹת שֶׁל סְפָרִים בְּכָל שִׁכְבוֹת הָעָם, בְּבָתֵּי כְּנֵסִיּוֹת וּבְבָתֵּי מִדְרָשׁוֹת וִישִׁיבוֹת וְכוּ', וְאֵיפֹה שֶׁרַק הָיָה יְהוּדִים הֵפַצְתִּי שָׁםכט.

הערות

  1. כטבְּרַם זָכוּר לַטּוֹב הָרַב הֶחָסִיד ר' צְבִי אַרְיֵה רוֹזֶנְפֶלְד זַ"ל, שֶׁשָּׁלַח בְּכָל יוֹם רִאשׁוֹן אַחַר הַצָּהֳרַיִם כַּמָּה בַּחוּרִים שֶׁהֶחְזִיר בִּתְשׁוּבָה, וְהָיִינוּ בְּיַחַד עֲשָׂרָה בְּמִסְפָּר, וְכָךְ הָלַכְנוּ מִבֵּית-הַכְּנֶסֶת לְבֵית-הַכְּנֶסֶת, וּמִבֵּית-הַמִּדְרָשׁ לְבֵית-הַמִּדְרָשׁ, וְהֵפַצְנוּ בְּיוֹם רִאשׁוֹן אֶחָד בְּיַחַד כְּאַלְפַּיִם סְפָרִים, וְזֶה הָיָה כָּל שָׁבוּעַ, וּכְשֶׁחָזְרוּ עָשִׂיתִי לָהֶם סְעֻדָּה גְּדוֹלָה, וְכָךְ הָיָה בְּמֶשֶׁךְ כָּל הַשָּׁנָה, וְלָכֵן לֹא בְּחִנָּם שֶׁקָּם עָלַי הַסָּמֶ"ךְ-מֵ"ם, כִּי דָּבָר זֶה עֲדַיִן לֹא הָיָה, לְהַדְפִּיס כָּל-כָּךְ הַרְבֵּה סְפָרִים בְּכַמֻיּוֹת כָּאֵלּוּ, וּלְמַלֵּא אֶת כָּל הַשְּׁכוּנוֹת הַחֲרֵדִיּוֹת, שֶׁאָז אֲפִלּוּ לֹא יָדְעוּ וְלֹא שָׁמְעוּ אֶת שְׁמוֹ הַקָּדוֹשׁ שֶׁל רַבֵּנוּ זַ"ל, זֶה הָיָה חִדּוּשׁ שֶׁבַּחִדּוּשִׁים. וְלָכֵן הַשָּׁנָה הַזּוֹ תשכ"ה עָבְרָה בְּהַצְלָחָה גְּדוֹלָה מְאֹד, וְהִנֵּה הִגִּיעָה שְׁנַת תשכ"ו, אָז הֵסִית אֶחָד אֶת הָרַב הֶחָסִיד ר' צְבִי אַרְיֵה רוֹזֶנְפֶלְד זַ"ל נֶגְדִּי, וּלְשׁוֹן-הָרַע מִתְקַבֶּלֶת, וְקָרָא אוֹתִי לְדִין תּוֹרָה, וְכַמּוּבָן שֶׁהוּא הִפְסִיד אֶת הַדִּין תּוֹרָה, וּמֵאָז הִתְחִיל פֵּירוּד בֵּינֵינוּ, אֲבָל הַקָּדוֹשׁ-בָּרוּךְ-הוּא שָׁמַר אוֹתִי מִכָּל רַע.
