סעיף טז

הִתְוַעֲדוּת לַסֵּפֶר הַקָּדוֹשׁ "מְשִׁיבַת נֶפֶשׁ", וּבְעֶזְרַת אָחִיו הַגָּאוֹן זַצַ"ל לְעוֹד סִפְרֵי רַבֵּנוּ, יְתָאֵר מַהוּת הַסְּפָרִים וְהַמּוֹצִיאִים לָאוֹר שֶׁלָּהֶם

תוכן

כְּשֶׁהִגַּעְתִּי לְגִיל חֲמֵשׁ-עֶשְׂרֵה וָחֵצִי, יוֹם אֶחָד מִהַרְתִּי מְאֹד לְדַרְכִּי אֶל הַיְּשִׁיבָה, וְלָכֵן תָּפַסְתִּי סֵפֶר קָטָן, וְהִכְנַסְתִּי אֶת זֶה בַּכִּיס, וְעַל הַדֶּרֶךְ הִתְחַלְתִּי לְעַיֵּן בָּזֶה, וְזֶה הָיָה סֵפֶר "מְשִׁיבַת נֶפֶשׁ"יג, וְהַסֵּפֶר מָשַׁךְ אֶת לְבָבִי, וְהִתְחַלְתִּי לִקְרֹא בַּסֵּפֶר עַמּוּד אַחַר עַמּוּד, וְאַף שֶׁהָיְתָה קָשָׁה הַקְּרִיאָה, כִּי הָיָה סֵפֶר יָשָׁן וְאוֹתִיּוֹת מְקֻטָּעוֹתיד, עִם כָּל זֹאת נִדְבַּקְתִּי אֶל הַסֵּפֶר, וְשָׁתִיתִי בַּצָּמָא כָּל עַמּוּד וְעַמּוּד, וּכְבָר לֹא עָזַבְתִּי אֶת הַסֵּפֶר עַד שֶׁגָּמַרְתִּי אֶת זֶה, וְהִתְחַלְתִּי אֶת זֶה עוֹד פַּעַם (בַּיּוֹם הַזֶּה כְּבָר לֹא לָמַדְתִּי כְּלוּם, כִּי כָּל-כָּךְ נִדְבַּקְתִּי בַּסֵּפֶר, עַד שֶׁלֹּא הִנַּחְתִּי אֶת זֶה מִיָּדִי, וְכָל הַיּוֹם עִיַּנְתִּי רַק בַּסֵּפֶר הַזֶּה, וְהִתְחַלְתִּי אֶת זֶה עוֹד פַּעַם, וְעַד שֶׁחָזַרְתִּי הַבַּיְתָה בַּלַּיְלָה לֹא הָלַכְתִּי לִישֹׁן, עַד שֶׁגָּמַרְתִּי בַּיּוֹם הַזֶּה אֶת הַסֵּפֶר שָׁלֹשׁ פְּעָמִים, וְכָל-כָּךְ נִדְבַּקְתִּי בְּהַסֵּפֶר, שֶׁאִי אֶפְשָׁר לְתָאֵר וּלְשַׁעֵר כְּלָל טו.

א. מִי אוֹמֵר שֶׁרַק אֵלּוּ יְכוֹלִים לִהְיוֹת צַדִּיקִים, אַדְרַבָּה כָּל בַּר יִשְׂרָאֵל שֶׁרַק רוֹצֶה, הוּא יָכוֹל לִהְיוֹת צַדִּיק וְדָבוּק בּוֹ יִתְבָּרַךְ. ב. מִי שֶׁמַּרְגִּיל אֶת עַצְמוֹ לְהִתְבּוֹדֵד עִמּוֹ יִתְבָּרַךְ, אֲזַי זוֹכֶה לִקְדֻשַּׁת וְטָהֳרַת הַמֹּחַ, עַד שֶׁזּוֹכֶה לְדַבֵּק אֶת עַצְמוֹ בּוֹ יִתְבָּרַךְ בְּיִחוּדִים וּבְשֵׁמוֹת וְכוּ' וְכוּ', וְלַעֲלוֹת לְמָקוֹם שֶׁיַּעֲלֶה.

הערות

  1. יגסֵפֶר זֶה הוּא אוֹת הִתְחַזְּקוּת מִלִּקּוּטֵי-עֵצוֹת הַמְשֻׁלָּשׁ, שֶׁהִדְפִּיס הָרַב הֶחָסִיד ר' אַלְטֶר טֶעפְּלִיקֶר זַ"ל, שֶׁמְּדַבֵּר רַק מֵהִתְחַזְּקוּת, וּמִגִּלּוּי אֱלקֹוּת לְכָל הַיְּרוּדִים וְהַפְּחוּתִים וְכוּ'.
  2. ידהַמַּהֲדוּרָה הַזֹּאת שֶׁל סֵפֶר "מְשִׁיבַת נֶפֶשׁ", הִדְפִּיס הָרַב הֶחָסִיד ר' צְבִי רַייכְמַן זַ"ל, תֵּכֶף-וּמִיָּד כְּשֶׁהִשְׁתַּחְרֵר מִמַּחֲנֶה הַהֶסְגֵּר בְּגֶרְמַנְיָה הָאֲרוּרָה, וְהִדְפִּיס אֶת זֶה בְּיַחַד בִּכְרִיכָה אַחַת עִם סֵפֶר "שִׂיחוֹת-הָרַ"ן", שֶׁזֶּה בִּכְלָל לֹא הָיוּ יְכוֹלִים לִקְרֹא, כִּי הָיָה אוֹתִיּוֹת קְטַנּוֹת וּמְקֻטָּעוֹת ]אַחַר הַרְבֵּה שָׁנִים פָּגַשְׁתִּי אוֹתוֹ, וְסִפֵּר לִי אֵיךְ שֶׁהַסֵּפֶר הַזֶּה לְרַבּוֹת הַסֵּפֶר "לִקּוּטֵי-מוֹהֲרַ"ן", הָיָה עִמּוֹ בְּמֶשֶׁךְ כָּל גָּלוּתוֹ בַּשּׁוֹאָה, וְזֶה אֲשֶׁר הֶחְזִיק אוֹתוֹ בְּמֶשֶׁךְ כָּל גָּלוּתוֹ, וְזֶה פֶּלֶא עַל גַּבֵּי פֶּלֶא אֵיךְ שֶׁהָיָה יָכוֹל לְהַחְבִּיא אֶת זֶה מֵהָרְשָׁעִים בְּמֶשֶׁךְ כָּל שְׁהוּתוֹ בְּמַחֲנֵה הַהֶסְגֵּר, וְלֹא מָצְאוּ אֶת זֶה אֶצְלוֹ, וְתֵכֶף-וּמִיָּד כְּשֶׁהִשְׁתַּחְרֵר, הַדָּבָר הָרִאשׁוֹן שֶׁדָּאַג, הָיָה לְהַדְפִּיס אֶת הַ"לִּקּוּטֵי-מוֹהֲרַ"ן", וְאֶת הַסֵּפֶר "מְשִׁיבַת נֶפֶשׁ" בְּיַחַד עִם הַסֵּפֶר "שִׂיחוֹת-הָרַ"ן".
  3. טווְיַעַן שֶׁבְּקִטְעֵי הַ"לִּקּוּטֵי-הֲלָכוֹת" שֶׁמּוּבָא בַּסֵּפֶר הַזֶּה, מוֹפִיעַ "בִּרְכוֹת הַשַּׁחַר" וְכוּ', "שַׁבָּת" "פֶּסַח" "רֹאשׁ הַשָּׁנָה" וְכוּ' וְכוּ', וְהָיָה אֶצְלִי פֶּלֶא אֵיךְ בַּשֻּׁלְחָן עָרוּךְ יֵשׁ הִתְחַזְּקוּת כָּזוֹ, וְלָקַחְתִּי שֻׁלְחָן עָרוּךְ וְחִפַּשְׂתִּי בְּהִלְכוֹת בִּרְכּוֹת הַשַּׁחַר, וּבְהִלְכוֹת שַׁבָּת, הֵן בַּמָּגֵן אַבְרָהָם וְהֵן בַּטּוּרֵי זָהָב, וְלֹא רָאִיתִי אֶת הַדְּבָרִים הָאֵלּוּ, וְהָיָה אֶצְלִי פֶּלֶא, וְשָׁאַלְתִּי אֶת אָחִי הַגָּדוֹל הרה"ג רַבִּי בִּנְיָמִין (שֶׁהָיָה בָּקִי נִפְלָא בְּכָל סֵפֶר שֶׁבָּעוֹלָם, מִי מְחַבְּרוֹ, וְכָל הַהִסְטוֹרְיָה שֶׁל הַבַּעַל מְחַבֵּר, וְלָכֵן שָׁאַלְתִּי אוֹתוֹ): לָמָּה אֲנִי לֹא מוֹצֵא אֶת הַדְּבָרִים הָאֵלּוּ בַּשֻּׁלְחָן עָרוּךְ? וְהֶרְאֵיתִי לוֹ אֶת הַקְּטָעִים שֶׁל "לִקּוּטֵי-הֲלָכוֹת", וְאָז צָחַק וְהִסְבִּיר לִי, שֶׁזֶּה לֹא כָּתוּב בַּשֻּׁלְחָן עָרוּךְ, אֶלָּא סִפֵּר לִי מֵרַבֵּנוּ זַ"ל וּמִתַּלְמִידוֹ מוֹהַרְנַ"תּ זַ"ל, וְזוֹ הַפַּעַם הָרִאשׁוֹנָה שֶׁשָּׁמַעְתִּי מֵהֶם. וְהִסְבִּיר לִי אֶת דַּרְכֵי הַ"לִּקּוּטֵי-הֲלָכוֹת", וְתֵכֶף-וּמִיָּד שָׁאַלְתִּי אוֹתוֹ: הַאִם יֵשׁ לְאַבָּא בַּסִּפְרִיָּה שֶׁלּוֹ עוֹד סְפָרִים מֵהַצַּדִּיקִים הָאֵלּוּ? וְתֵכֶף-וּמִיָּד נָתַן לִי שְׁנֵי "לִקּוּטֵי-מוֹהֲרַ"ן" שֶׁהָיָה בְּבֵיתֵנוּ (אֶחָד שֶׁהִדְפִּיס הָרַב הֶחָסִיד ר' שְׁמוּאֵל הוֹרְבִיץ זַ"ל בִּשְׁנַת תרצ"ה, וְאֶחָד שֶׁהִדְפִּיס הָרַב הֶחָסִיד ר' צְבִי רַייכְמַן זַ"ל כַּנַּ"ל), וְעוֹד נָתַן לִי "סֵפֶר-הַמִּדּוֹת", וְכֵן סֵפֶר "הִשְׁתַּפְּכוּת הַנֶּפֶשׁ", שֶׁהָיוּ בִּכְרִיכָה רַכָּה, שֶׁנִּדְפְּסוּ בְּאַרְצוֹת-הַבְּרִית בִּשְׁנַת תש"ט, וְאֵיךְ שֶׁהָיוּ בְּיָדִי סְפָרִים אֵלּוּ, הָיְתָה שִׂמְחָתִי לְאֵין קֵץ, וְהִתְחַלְתִּי לְעַיֵּן בְּ"לִקּוּטֵי-מוֹהֲרַ"ן", וְלֹא הֵבַנְתִּי שׁוּם דָּבָר, וְהָיָה אֶצְלִי הַסֵּפֶר כְּסֵפֶר חָתוּם, וְלָכֵן הִסְתַּכַּלְתִּי הַרְבֵּה בְּ"סֵפֶר-הַמִּדּוֹת", וְהָעִקָּר בַּסֵּפֶר "הִשְׁתַּפְּכוּת הַנֶּפֶשׁ" וְאֵיךְ שֶׁעִיַּנְתִּי הַרְבֵּה מְאֹד בַּסֵּפֶר "הִשְׁתַּפְּכוּת הַנֶּפֶשׁ", נִפְתַּח לְפָנַי עוֹלָם חָדָשׁ, וְנִתְגַּלָּה לִי כָּל הָעִנְיָן שֶׁל הִתְבּוֹדְדוּת וְשִׂיחָה בֵּינוֹ לְבֵין קוֹנוֹ, וְתֵכֶף-וּמִיָּד תֹּרְצוּ לִי כָּל הַקֻּשְׁיוֹת:
