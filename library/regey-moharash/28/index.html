סעיף כח

נִתְגַּלָּה סוֹד הִתְקָרְבוּתוֹ וּגְדֻלַּת הָרַב הֶחָסִיד ר' יְחֶזְקֵאל מַרְגָּלִיּוֹת זַ"ל מֵאַנְשֵׁי שְׁלוֹמֵנוּ

תוכן

עַד שֶׁיּוֹם אֶחָד נִתְגַּלָּה לְכֻלָּם הַסּוֹד, שֶׁאֲנִי מְקֹרָב אֶל חֲסִידוּת בְּרֶסְלֶב ]כִּי בֶּאֱמֶת כָּל הַזְּמַן הַחֲבֵרִים שֶׁלִּי רָצוּ לָדַעַת לָמָּה הִשְׁתַּנֵּיתִי, כִּי הָיִיתִי מְאֹד חֶבְרוּתִי וְרֹאשׁ הַמְדַבְּרִים, וְכֻלָּם הָיוּ חֲבֵרַי, וּפִתְאֹם כָּל-כָּךְ הִשְׁתַּנֵּיתִי מֵהֵפֶךְ אֶל הֵפֶךְ, עַד שֶׁחָשְׁבוּ שֶׁנִּסְגַּרְתִּי לְגַמְרֵי, כִּי לֹא יָדְעוּ מַה שֶּׁאֲנִי עוֹשֶׂה בַּסֵּתֶר, וְלָכֵן הָיְתָה לָהֶם פְּלִיאָה מְאֹד, עַד שֶׁנִּתְגַּלָּה לְכֻלָּם, וְאָז הִתְחִילָה עָלַי הַמַּחֲלֹקֶת וְהַלֵּיצָנוּת וְהָרְדִיפוֹת בַּיְשִׁיבָה, וְהָרֹאשׁ יְשִׁיבָה קָרָא לְאָבִי, וְהִתְחַלְתִּי לִטְעֹם מְרִירוּתכא[, כִּי הַמַּשְׁגִּיחַ הִתְחִיל לְכַנּוֹת אוֹתִי בְּשֵׁמוֹת שֶׁל גְּנַאי, וּבְפֵרוּשׁ הוֹרָה לַבַּחוּרִים, שֶׁלֹּא יִהְיֶה לָהֶם עֵסֶק עִמִּי, וְצִוָּה לְכֻלָּם, שֶׁלֹּא יִסְתַּכְּלוּ בְּסִפְרֵי רַבֵּנוּ זַ"ל, כִּי מִי שֶׁמִּסְתַּכֵּל בָּהֶם - מִשְׁתַּגֵּעַ וְכוּ' וְכוּ', כָּל מִינֵי הֲסָתוֹת הֵסִית נֶגֶד סִפְרֵי רַבֵּנוּ זַ"ל, וְסָבַלְתִּי מִזֶּה הַרְבֵּה מְאֹד, וְיַעַן שֶׁהָיִיתִי עַקְשָׁן, הֵבֵאתִי סִפְרֵי "מְשִׁיבַת נֶפֶשׁ" וְ"הִשְׁתַּפְּכוּת הַנֶּפֶשׁ", וְהֵפַצְתִּי אֶת זֶה בֵּין הַבַּחוּרִים, וּלְבַסּוֹף אַחַר צְעָקוֹת וּבִזְיוֹנוֹת וְעֶלְבּוֹנוֹת שֶׁעָשָׂה לִי, רִחֵק אוֹתִי מֵהַיְשִׁיבָה, וְאַף שֶׁלָּמַדְתִּי שָׁם חָמֵשׁ שָׁנִים, הֻצְרַכְתִּי לְחַפֵּשׂ יְשִׁיבָה אַחֶרֶת.

וְלָכֵן כְּשֶׁרָאָה פַּעַם שֶׁאֲנִי מְעַיֵּן בַּסֵּפֶר "עָלִים לִתְרוּפָה", הִתְחִיל לִצְעֹק לְכֻלָּם, הִנֵּה הוּא נַעֲשָׂה בְּרֶסְלֶבֶר, וְכָךְ נִתְפַּרְסֵם בֵּין כָּל הַבַּחוּרִים, וְאַף שֶׁבִּקַּשְׁתִּי אוֹתוֹ לִהְיוֹת בְּשֶׁקֶט, וְלֹא לְגַלּוֹת לְאַף אֶחָד, הוּא בְּשֶׁלּוֹ, גִּלָּה אוֹתִי לְכָל הַחֲבֵרִים, וְאִם כִּי מִצַּד אֶחָד הָיָה לִי צַעַר גָּדוֹל מְאֹד - לָמָּה הָיָה צָרִיךְ לְפַרְסֵם אוֹתִי, כִּי הִתְחִילוּ לַעֲקבֹ אַחֲרַי מְאֹד אֵיפהֹ אֲנִי נֶעְלָם וְכוּ', (מַה שֶּׁבֶּאֱמֶת לֹא הִצְלִיחוּ, כִּי הָיִיתִי יוֹתֵר חָכָם מֵהֶם, וּבְכָל פַּעַם עָשִׂיתִי בְּחָכְמָה אַחֶרֶת אֵיךְ לְהֵעָלֵם מֵהֶם, שֶׁלֹּא יֵדְעוּ אֵיפהֹ אֲנִי הוֹלֵךְ), אֲבָל מִצַּד שֵׁנִי שָׂמַחְתִּי, שֶׁמֵעַכְשָׁו אוּכַל לְהָפִיץ בֵּינֵיהֶם גַּם-כֵּן אֶת סִפְרֵי רַבֵּנוּ זַ"ל, כִּי זֶה הָיָה חֶשְׁקִי וּרְצוֹנִי חָזָק מְאֹד, לָמָּה שֶׁעוֹד בַּחוּרִים לֹא יֵדְעוּ מֵרַבֵּנוּ זַ"ל?! וְכָךְ לְאַט לְאַט הִתְחַלְתִּי לְדַבֵּר עִם הַבַּחוּרִים מֵעִנְיְנֵי רַבֵּנוּ זַ"ל, וְגַם הֵבֵאתִי כַּמָּה סְפָרִים, וְאָז הִתְחִילָה עָלַי הַמַּחֲלֹקֶת מֵרֹאשׁ הַיְשִׁיבָה, שֶׁהִתְחִיל לִרְדֹּף אוֹתִי מְאֹד מְאֹד.

הערות

  1. כאוְהַסִּבָּה אֵיךְ שֶׁנִּתְגַּלָּה, הָיָה עַל-יְדֵי בָּחוּר אֶחָד. וּמַעֲשֶׂה שֶׁהָיָה כָּךְ הָיָה, הַבָּחוּר הַזֶּה נָסַע לְאַנְגְלִיָּה בִּשְׁבִיל שִׁדּוּךְ, וְהַשִּׁדּוּךְ לֹא יָצָא מִכֹּחַ אֶל הַפֹּעַל, וְהָיָה מְאֹד שָׁבוּר, מַמָּשׁ חָשַׁב שֶׁזֶּה סוֹף הָעוֹלָם, כִּי נָסַע מֵאַרְצוֹת-הַבְּרִית עַד אַנְגְלִיָּה לְשִׁדּוּךְ, וְלֹא יָצָא מִזֶּה כְּלוּם, אָז לֹא הָיָה יָכוֹל לְקָרֵר אֶת עַצְמוֹ מֵרֹב צַעַר, וְהִתְאַכְסֵן בְּאֵיזוֹ אַכְסַנְיָה שֶׁל הַכְנָסַת אוֹרְחִים, וְלֹא הָיָה יָכוֹל לִישֹׁן מֵרֹב עָגְמַת נֶפֶשׁ וַחֲלִישׁוּת הַדַּעַת. וְהִנֵּה בְּאוֹתוֹ חֶדֶר שֶׁהוּא יָשַׁן, הָיָה יְהוּדִי מְשֻׁלָּח מֵאֶרֶץ יִשְׂרָאֵל, בְּשֵׁם הָרַב הֶחָסִיד ר' יְחֶזְקֵאל מַרְגָּלִיּוֹת, זִכְרוֹנוֹ לִבְרָכָה (וְהוּא הָיָה אֶחָד מֵחֲשׁוּבֵי אַנְשֵׁי שְׁלוֹמֵנוּ), וְהַבָּחוּר הַזֶּה לֹא יָדַע מִי זֶה, אֶלָּא הוּא רָאָה שֶׁהוּא בָּא מְאֹד מְאֻחָר כָּל לַיְלָה, וְהוּא מִסְתַּכֵּל אִם אֵינוֹ יָשֵׁן, וּבְדֶרֶךְ כְּלָל הוּא כְּבָר הָיָה יָשֵׁן, אֲבָל בְּאוֹתוֹ לַיְלָה שֶׁהָיְתָה לוֹ פְּגִישָׁה, וְלֹא יָצָא מֵהַשִּׁדּוּךְ כְּלוּם, מֵרֹב צַעַר וְעָגְמַת נֶפֶשׁ לֹא הָיָה יָכוֹל לִישֹׁן, וּכְשֶׁבָּא הַנַּ"ל לְתוֹךְ הַחֶדֶר, עָשָׂה אֶת עַצְמוֹ כְּאִלּוּ הוּא יָשֵׁן, וְהַלָּה נִגַּשׁ לִרְאוֹת אִם הוּא יָשֵׁן, וֶהֱיוֹת שֶׁעֵינָיו הָיוּ עֲצוּמוֹת, חָשַׁב שֶׁהוּא יָשֵׁן, וְהִנֵּה הַיְּהוּדִי הַנַּ"ל הִדְלִיק נֵר וְהוֹצִיא סֵפֶר מִתִּיקוֹ, וְהִתְיַשֵּׁב עַל הָרִצְפָּה וּבָכָה בְּכִיּוֹת עֲצוּמוֹת, וְהוּא רַק מִסְתַּכֵּל עָלָיו, וְהַלָּה חָשַׁב שֶׁהוּא יָשֵׁן, וְאַחַר-כָּךְ הוֹצִיא עוֹד סֵפֶר גָּדוֹל מִתִּיקוֹ, וְכָךְ יָשַׁב כַּמָּה שָׁעוֹת וּבָכָה, שֶׁעוֹד לֹא רָאָה אַף פַּעַם שֶׁאָדָם יִבְכֶּה כָּל-כָּךְ הַרְבֵּה, וְקֹדֶם אוֹר הַיּוֹם קָם מֵהָרִצְפָּה, וְהִכְנִיס אֶת הַסֵּפֶר לְתִיקוֹ, וְלָקַח עִמּוֹ אֶת הַטַּלִּית וּתְפִלִּין וְיָצָא, וְאָז הַבָּחוּר לֹא הָיָה יָכוֹל לְהִתְאַפֵּק, וְהָיָה לוֹ פֶּלֶא אֵיזֶה סֵפֶר זֶה יָכוֹל לִהְיוֹת שֶׁאָדָם יַגִּיד בָּזֶה כָּל-כָּךְ הַרְבֵּה זְמַן, וְיִבְכֶּה בְּכִיּוֹת עֲצוּמוֹת, וְלָכֵן נִגַּשׁ אֶל הַתִּיק שֶׁלּוֹ, וּפָתַח אֶת זֶה, וּמָצָא שָׁם סֵפֶר "שַׁעֲרֵי צִיּוֹן", וְהַסֵּפֶר הַגָּדוֹל הָיָה סֵפֶר "לִקּוּטֵי-תְּפִלּוֹת", וְרָאָה אֵיךְ שֶׁהַדַּפִּים שֶׁל הַסֵּפֶר הָיוּ רְטֻבִּים מְאֹד מִדְּמָעוֹת, וְהִתְחִיל לְעַיֵּן בַּסֵּפֶר, וּמֵחֲמַת שֶׁהוּא בְּעַצְמוֹ הָיָה שָׁבוּר, הִתְחִיל לוֹמַר תְּפִלּוֹת מִ"לִּקּוּטֵי תְּפִלּוֹת", וְזֶה מָשַׁךְ אֶת לִבּוֹ מְאֹד מְאֹד, עַד שֶׁנִּסְכַּם אֶצְלוֹ, שֶׁהַיְּהוּדִי הַזֶּה מֻכְרָח לִהְיוֹת אֵיזֶה צַדִּיק נִסְתָּר, מִל"ו צַדִּיקִים, כִּי אַף פַּעַם בַּחַיִּים לֹא רָאָה אָדָם שֶׁיִּבְכֶּה כָּל-כָּךְ הַרְבֵּה בְּלַיְלָה, וְהֵבִין שֶׁהָאָדָם הַזֶּה אָמַר תִּקּוּן חֲצוֹת, וְאַחַר-כָּךְ שָׁעוֹת הָיָה שׁוֹפֵךְ דְּמָעוֹת כַּמַּיִם בַּסֵּפֶר "לִקּוּטֵי-תְּפִלּוֹת", וּבַבֹּקֶר הַבָּחוּר קָם וְהָלַךְ לְהִתְפַּלֵּל, וְהָיָה לוֹ רָצוֹן גָּדוֹל לְהִפָּגֵשׁ עִם הַיְּהוּדִי הַזֶּה, כְּדֵי לִתְפֹּס אִתּוֹ שִׂיחָה, וְכֵן הָיָה שֶׁחָזַר בַּצָּהֳרַיִם לֶאֱכלֹ, אָז הָיְתָה לוֹ הִזְדַּמְּנוּת לְדַבֵּר עִמּוֹ, וְהַבָּחוּר הִתְחִיל לִשְׁפֹּךְ אֶת לִבּוֹ הַמַּר, אֵיךְ שֶׁבָּא לְפֹה בְּעִנְיַן שִׁדּוּךְ,
