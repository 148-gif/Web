סעיף כט

רֹאשׁ הַשָּׁנָה תש"כ בִּמְחִיצַת אֲנַ"שׁ הַתְּמִימִים, לִינָה בִּישִׁיבַת תִּפְאֶרֶת-יְרוּשָׁלַיִם שֶׁל הגר"מ פַיְינְשְׁטֵיין זי"ע וְלִמּוּד שָׁם בְּמֶשֶׁךְ כַּמָּה שָׁנִים, וְהִתְקָרְבוּת מְיֻחֶדֶת לְהַר' וֶועלְוְול חֶשִׁין זי"ע

תוכן

וְיַעַן שֶׁהָיְתָה שְׁכוּנָה בִּנְיוּ-יוֹרְק, שֶׁשָּׁם הָיָה הַשְּׁטִיבְּל הַיְּחִידִי שֶׁל חֲסִידֵי בְּרֶסְלֶב בְּכָל אַרְצוֹת- הַבְּרִית, וְשָׁם נִתְקַבְּצוּ עַל רֹאשׁ הַשָּׁנָה, וְלָכֵן בַּזְּמַן הֶחָדָשׁ רָצִיתִי לָבוֹא עַל רֹאשׁ הַשָּׁנָה, וְלֹא הָיָה לִי אֵיפהֹ לְהִתְאַכְסֵן, וְיַעַן שֶׁלֹּא רָחוֹק מִשָּׁם הָיְתָה יְשִׁיבַת "תִּפְאֶרֶת יְרוּשָׁלַיִם", תַּחַת הַנְהָגָתוֹ שֶׁל הַגָּאוֹן הָאַדִּיר שַׂר הַתּוֹרָה ר' מֹשֶׁה פַיְינְשְׁטֵין זְצַ"ל, אָמַרְתִּי לְעַצְמִי שֶׁאֲנִי אֶתְאַכְסֵן שָׁמָּה, וְיַעַן שֶׁהָיָה לִי שָׁם יָדִיד שֶׁלָּמַד שָׁם, הוּא סִדֵּר לִי מְקוֹם לִינָה, וְלָכֵן בְּרֹאשׁ הַשָּׁנָה תש"ככב הִתְאַכְסַנְתִּי שָׁמָּה בַּיְשִׁיבָה, וּמְאֹד מָצָא חֵן בְּעֵינַי הַמָּקוֹם, כִּי סַךְ הַכּלֹ לָמְדוּ שָׁם כַּחֲמִשָּׁה-עָשָׂר בַּחוּרִים מְבֻגָּרִים, וְהֵבַנְתִּי שֶׁפּהֹ טוֹב לִי, כִּי אֲנִי יָכוֹל לִלְמֹד מַה שֶּׁאֲנִי רוֹצֶה, וְתֵכֶף-וּמִיָּד אַחַר רֹאשׁ הַשָּׁנָה הִתְחַלְתִּי לָבֹא לְהַיְשִׁיבָה, וְרֹאשׁ הַיְשִׁיבָה קִבֵּל אוֹתִי בְּכָבוֹד גָּדוֹל, כִּי הָיִיתִי שָׁם מַתְמִיד גָּדוֹל מְאֹד, וְזֶה מָצָא חֵן בְּעֵינֵי הָרֹאשׁ יְשִׁיבָה, שֶׁהָיָה גָּאוֹן מְפֻרְסָם מְאֹד, וְהִנֵּה רוֹאֶה שֶׁבָּחוּר חֲסִידִי יוֹשֵׁב לוֹ בַּפִּנָּה עִם סְטֶנְדֵּר, וְלוֹמֵד כְּסִדְרָן שַׁ"ס, כִּי אָז הִשְׁקַעְתִּי הַרְבֵּה בְּלִמּוּד כְּסִדְרָן, וְהָיִיתִי לוֹמֵד בְּכָל יוֹם הַרְבֵּה דַּפִּים גְּמָרָא כְּסִדְרָן, בְּחִיּוּת נִפְלָאָה, כִּי לֹא הָיָה לִי מִי שֶׁיַּפְרִיעַ לִי, וְכֵן לָמַדְתִּי מִשְׁנָיוֹת, וְהָרֹאשׁ יְשִׁיבָה רָאָה, שֶׁיּוֹשֵׁב פּהֹ בָּחוּר מַתְמִיד, שֶׁיָּכוֹל לָשֶׁבֶת שָׁעוֹת עַל גַּבֵּי שָׁעוֹת בְּמָקוֹם אֶחָד וְלֹא לָזוּז וְכוּ', אָז הוּא מְאֹד מְאֹד כִּבֵּד אוֹתִי, וְקֵרֵב אוֹתִי בְּכָל מִינֵי הִתְקָרְבוּת, עַד כְּדֵי כָּךְ, שֶׁאֲנִי הָיִיתִי כָּל-כָּךְ שָׁקוּעַ בְּלִמּוּד, וְהוּא הָיָה עוֹבֵר אֵיפהֹ שֶׁאֲנִי יוֹשֵׁב, וְהִסְתַּכֵּל בְּכָל פַּעַם אֵיזֶה דַּף אֲנִי כְּבָר אוֹחֵז, וְהָיָה נוֹתֵן לִי לִטּוּף עַל הַגַּב פִּתְאֹם, וְכָךְ הִפְסַקְתִּי כְּדֵי לְהִסְתַּכֵּל מִי מְלַטֵּף אוֹתִי?! וְהוּא בְּרֹב עַנְוְתָנוּתוֹ אָמַר לִי: לֶערְין לֶערְין ]תִּלְמַד, תִּלְמַד[, וְנֶהֱנָה מְאֹד מְאֹד מִמֶּנִּי, וְלָכֵן הָיְתָה לִי כְּבָר שָׁם מְנוּחָה.

וּבְעִנְיַן הַסִּפְרִיָּה הַזּוֹ, יֵשׁ סִפּוּר אָרֹךְ, כִּי מֵאָז שֶׁהִתְמַדְתִּי בְּסִפְרֵי רַבֵּנוּ זַ"ל, כְּשֶׁבָּאתִי לְמָקוֹם שֶׁרַבֵּנוּ זַ"ל הִזְכִּיר אֵיזֶה סֵפֶר, רָצִיתִי מְאֹד לְהַשִּׂיג אֶת הַסֵּפֶר, וְלֹא הָיָה בְּנִמְצָא, כְּגוֹן: כְּשֶׁבָּאתִי לְחַיֵּי-מוֹהֲרַ"ן, סִימָן שס"ד, שֶׁדִּבֵּר רַבֵּנוּ זַ"ל, שֶׁקַּבָּלַת הָרַמָּ"ק וְקַבָּלַת הָאֲרִיזַ"ל הֵם שְׁנֵי דְּבָרִים נִפְרָדִים, אֲזַי נִכְנָס בִּי רָצוֹן וְחֵשֶׁק גָּדוֹל לִלְמֹד, וּלְעַיֵּן בְּסִפְרֵי הָרַמָּ"ק, כְּגוֹן הָאֵלִימָה, וְהַפַּרְדֵּס רִמּוֹנִים וְכוּ', וְכֵן כִּתְבֵי הָאֲרִיזַ"ל, שֶׁלֹּא הָיוּ מְצוּיִים אָז בִּכְלָל, וְלָכֵן חִפַּשְׂתִּי סִפְרִיָּה אֵיפהֹ שֶׁאֲנִי יָכוֹל לְהַשִּׂיג אֶת כָּל הַסְּפָרִים הַיְשָׁנִים, וְסִבֵּב הַקָּדוֹשׁ-בָּרוּךְ-הוּא, שֶׁהִגַּעְתִּי לַסִּפְרִיָּה הֲכִי גְּדוֹלָה שֶׁיֵּשׁ שָׁם כָּל סִפְרֵי קֹדֶשׁ, שֶׁבִּכְלָל אֵינָם נִמְצָאִים, וּמָצָאתִי חֵן בְּעֵינֵי הַמְּמֻנִּים, וְנָתְנוּ לִי כַּרְטִיס חָבֵר (וְאַף שֶׁזֶּה לֹא הָיָה מָקוֹם לְבָחוּר חֲסִידִי לָלֶכֶת שָׁמָּה, עִם כָּל זֹאת מֵרֹב תְּשׁוּקָה לְחַטֵּט וּלְחַפֵּשׂ בִּסְפָרִים יְשָׁנִים לֹא הִסְתַּכַּלְתִּי עַל שׁוּם דָּבָר), וְכָךְ שָׁאַלְתִּי מִשָּׁם אֶת סִפְרֵי הַפַּרְדֵּס רִמּוֹנִים, וְהָאֵלִימָה וְכוּ', וְשָׁקַדְתִּי בָּהֶם מְאֹד, וּבְכָל פַּעַם שָׁאַלְתִּי מִשָּׁם סְפָרִים יְשָׁנִים, וְזֶה הֶחֱיָה אוֹתִי מְאֹד מְאֹד, עַד שֶׁאַחַר חֲתֻנָּתִי כְּבָר הִדְפִּיסוּ אֶת הַסְּפָרִים וְקָנִיתִי אוֹתָם, עִם כָּל זֹאת בִּלִּיתִי הַרְבֵּה בַּסִּפְרִיָּה בְּעֵת בַּחֲרוּתִי, וּמָצָאתִי שָׁם סְפָרִים יְשָׁנִים, סְפָרִים יִקְרֵי הַמְּצִיאוּת, וְהֶעְתַּקְתִּי מֵהֶם הַרְבֵּה דְּבָרִים נִפְלָאִים, וּבָזֶה הֶחֱיֵיתִי אֶת עַצְמִי, וְאַף בִּרְיָה לֹא יָדְעָה אֵיפהֹ הָלַכְתִּי, וְיָשַׁבְתִּי בַּפִּנָּה, וְהֶעְתַּקְתִּי מִסְּפָרִים שֶׁלֹּא נָתְנוּ לְהוֹצִיא, מֵרֹב יִקְרַת הַמְּצִיאוּת שֶׁל הַסֵּפֶר.

הערות

  1. כבוְזוֹ הָיְתָה הַפַּעַם הָרִאשׁוֹנָה בְּרֹאשׁ הַשָּׁנָה תש"כ, שֶׁהִתְפַּלַּלְתִּי עִם הַקִּבּוּץ שֶׁל אַנְשֵׁי שְׁלוֹמֵנוּ, וְהָיָה שָׁם סָךְ הַכֹּל כַּחֲמִשָּׁה- עָשָׂר אֲנָשִׁים, שֶׁבָּאוּ מִכָּל חֶלְקֵי אַרְצוֹת-הַבְּרִית, וְגַם שְׁנֵי אַנְשֵׁי שְׁלוֹמֵנוּ מִטּוֹרוֹנְטוֹ, כִּי דְּבַר רַבֵּנוּ זַ"ל בִּכְלָל לֹא הָיָה מְפֻרְסָם אָז בְּאַרְצוֹת-הַבְּרִית, וּבְרֹאשׁ הַשָּׁנָה הַזּוֹ הִתְפַּלֵּל עִמָּנוּ הָרַב הֶחָסִיד ר' בִּנְיָמִין זְאֵב חֵשִׁין, זֵכֶר צַדִּיק לִבְרָכָה, וְנִתְקַשַּׁרְתִּי אֵלָיו מְאֹד, כִּי הוּא הָיָה חִדּוּשׁ גָּדוֹל, עַד שֶׁאֲפִלּוּ אַנְשֵׁי שְׁלוֹמֵנוּ בִּירוּשָׁלַיִם לֹא הִכִּירוּ אֶת גְּדֻלָּתוֹ, וַאֲנִי יָכוֹל לְהָעִיד, שֶׁהָאָדָם הַזֶּה הָיָה בָּקִי בַּזֹּהַר כְּמוֹ שֶׁיְּהוּדִי בָּקִי בְּ'אַשְׁרֵי', בְּלִי שׁוּם גֻּזְמָא כְּלָל, כִּי מֵחֲמַת שֶׁבַּסֵּתֶר לָמַדְתִּי הַרְבֵּה זֹהַר, וּכְשֶׁדִּבַּרְתִּי עִמּוֹ רָאִיתִי שֶׁתָּמִיד מַזְכִּיר אֶת הַזֹּהַר, וְלָכֵן הִתְחַלְתִּי לִשְׁאלֹ אוֹתוֹ אֵיפֹה כָּתוּב עִנְיָן זֶה בַּזֹּהַר וְכוּ', וְאֵיפֹה כָּתוּב עִנְיָן זֶה בַּזֹּהַר, וְהוּא עָנָה לִי מַמָּשׁ עַל הַמָּקוֹם, כְּאִלּוּ שֶׁרַק עַכְשָׁו לָמַד אֶת זֶה, וּמְאֹד הִתְפַּעַלְתִּי מִמֶּנּוּ, כִּי עוֹד לֹא רָאִיתִי אָדָם, שֶׁיִּהְיֶה כָּל-כָּךְ בָּקִי כָּמוֹהוּ, וּמַה גַּם אַחַר רֹאשׁ הַשָּׁנָה הוּא בָּא אֵלַי לַיְשִׁיבָה, וּבִקֵּשׁ אוֹתִי, יַעַן שֶׁלֹּא מַכִּיר אֶת נְיוּ-יוֹרְק, שֶׁאֲנִי אֵלֵךְ אִתּוֹ וְאַרְאֶה לוֹ אֵיפהֹ לָלֶכֶת, כִּי הָיוּ לוֹ הֲמוֹן כְּתוֹבוֹת, אֲבָל לֹא יָדַע אֵיפֹה, וְרָצָה שֶׁאֲנִי אֵלֵךְ אִתּוֹ, אֲבָל הֱיוֹת שֶׁהָיִיתִי מַתְמִיד גָּדוֹל מְאֹד, וְלֹא רָצִיתִי לְהַפְסִיק מִלִּלְמֹד, הִתְחַמַּקְתִּי מִמֶּנּוּ בְּתֵרוּצִים שׁוֹנִים, עִם כָּל זֹאת הָיְתָה לִי מְשִׁיכָה מְאֹד אַחֲרָיו, כִּי מְאֹד נֶהֱנֵיתִי, שֶׁאָדָם יוֹדֵעַ אֶת הַזֹּהַר בְּעַל-פֶּה, וְיַעַן שֶׁהִבְחַנְתִּי שֶׁעִם מִי שֶׁדִּבֵּר, הַדָּבָר הָרִאשׁוֹן הוּא בִּקֵּשׁ אֶת הַיָּד, הָיָה אוֹמֵר: תֵּן לִי אֶת יָדְךָ לְרֶגַע, וְהוּא הָיָה מְעַיֵּן בַּיָּד שֶׁל בְּנֵי אָדָם, וְהָיָה עוֹשֶׂה פַּרְצוּפִים שׁוֹנִים וְכוּ', וְתֵכֶף-וּמִיָּד שָׁאַלְתִּי אוֹתוֹ: הַאִם אַתֶּם בְּקִיאִים בְּחָכְמַת הַיָּד הַמּוּבָא בַּזֹּהַר יִתְרוֹ? אֲזַי הוּא מְאֹד הִתְפַּלֵּא מֵאֵיפֹה אֲנִי יוֹדֵעַ מִזֹּהַר כָּזֶה?! וּבֵין כָּךְ הִתְחִיל לְדַבֵּר אִתִּי מֵחָכְמָה זוֹ, אֲזַי עָשִׂיתִי אִתּוֹ תְּנַאי, שֶׁאֲנִי הוֹלֵךְ אִתּוֹ רַק בִּתְנַאי שֶׁיִּלְמַד אִתִּי אֶת חָכְמַת הַיָּד, וּבִתְחִלָּה הוּא לֹא רָצָה, אָז אָמַרְתִּי לוֹ שֶׁאֲנִי לֹא רוֹצֶה לָלֶכֶת עִמּוֹ, וֶהֱיוֹת שֶׁהָיָה צָרִיךְ אוֹתִי, הִסְכִּים לְלַמֵּד אוֹתִי, וְלִמֵּד אוֹתִי אֶת הַחָכְמָה הַזּוֹ, וּמְאֹד מְאֹד הִתְפַּעַלְתִּי מִמֶּנּוּ עוֹד יוֹתֵר, וְהָלַכְתִּי עִמּוֹ לְהַרְבֵּה אֲנָשִׁים, וְרָאִיתִי אֶצְלוֹ הַרְבֵּה דְּבָרִים נִפְלָאִים, פִּלְאֵי פְּלָאִים, וְאָז רָאִיתִי אֶל מָה אָדָם יָכוֹל לִזְכּוֹת, וְאַף שֶׁאֲנִי הָיִיתִי בָּחוּר צָעִיר, אֲבָל הָיְתָה לִי תְּשׁוּקָה עֲצוּמָה לַעֲבוֹדַת הַשֵּׁם, וְרָאִיתִי אֵצֶל הָאָדָם הַזֶּה דְּבָרִים מְאֹד מְאֹד מְעַנְיְנִים (וּבֵין כָּךְ כְּשֶׁלִּמֵּד אוֹתִי אֶת חָכְמַת הַיָּד הָלַכְתִּי לַסִּפְרִיָּה הֲכִי גְּדוֹלָה שֶׁהָיְתָה בִּנְיוּ-יוֹרְק, שֶׁהָיוּ לָהֶם הֲכִי הַרְבֵּה סְפָרִים מֵהַיַּהֲדוּת שֶׁיֵּשׁ בָּעוֹלָם, וְשָׁם מָצָאתִי עוֹד כַּמָּה סְפָרִים מְאֹד יְשָׁנִים הַמְדַבְּרִים מֵחָכְמָה זוֹ, וְהֶעְתַּקְתִּי אֶת זֶה, וְאַחַר-כָּךְ הֵבֵאתִי לוֹ אֶת זֶה, וְהוּא אַף פַּעַם לֹא רָאָה אֶת הַסְּפָרִים הָאֵלּוּ, וּמְאֹד מְאֹד שָׂמַח וְאָמַר לִי: "הֲרֵי זֶה בָּא לְלַמֵּד, וְנִמְצָא לָמֵד", וְנֶהֱנָה מְאֹד מְאֹד מֵהַסְּפָרִים שֶׁהֶעֱתַקְתִּי בִּכְתָב יָדִי, וְגַם הֶעְתַּקְתִּי מִכִּתְבֵי יָדוֹת שֶׁמָּצָאתִי שָׁם הַמְדַבְּרִים מֵעִנְיָנִים אֵלּוּ, אַךְ הוּא הִזְהִיר אוֹתִי שֶׁלֹּא אֲגַלֶּה לְאַף אֶחָד שֶׁלִּמֵּד אוֹתִי אֶת הַחָכְמָה הַזּוֹ, אֲבָל אֲנִי מֵרֹב תְּשׁוּקָה, כְּבָר הָיוּ לִי הַרְבֵּה סְפָרִים מֻעְתָּקִים מִכִּתְבֵי יָד שֶׁהֶעְתַּקְתִּי, וְגַם לִמֵּד אוֹתִי אֶת הַיְסוֹדוֹת בְּחָכְמָה זוֹ, וּכְבָר לֹא הֻצְרַכְתִּי יוֹתֵר לִלְמֹד אֵצֶל אַף אֶחָד).
